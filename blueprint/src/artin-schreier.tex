\section{Cyclic Artin--Schreier extensions of the Laurent field}

Throughout this section, $p$ is prime, $k=\mathbf F_p$, and
$K=k((t))$. Write $v$ for the additive order valuation, normalized by
$v(t)=1$ and $v(0)=+\infty$, and write $\wp(x)=x^p-x$.
The following construction supplies the cyclic case of the
Artin--Schreier correspondence used in \cite{Camina1997}, Theorem~1.
It proves the converse of the ramification note following her Lemma~3:
a reduced parameter with a negative pole gives a totally ramified extension.
The weighted valuation, completeness, and Laurent expansion proofs below
supply details beyond the cited text. The valuation is constructed on a
polynomial basis; existence of an extension of $v$ is not assumed.

\begin{lemma}[A pole prime to the characteristic is not an Artin--Schreier image]
\label{lem:artin-schreier-no-root-from-pole}
\source{camina1997:page-0005}
\lean{LaurentSeries.pow_sub_ne_of_orderTop}
\leanok
If $\gamma\in K$, $m>0$, $p\nmid m$, and $v(\gamma)=-m$, then
$\gamma\notin\wp(K)$.
\end{lemma}

\begin{proof}
Suppose $\gamma=x^p-x$. If $v(x)\ge0$, then
$v(x^p-x)\ge\min\{p v(x),v(x)\}\ge0$, a contradiction.
Thus $v(x)<0$. Since $p\ge2$, we have $p v(x)<v(x)$, so the
two summands have different valuations and
$v(x^p-x)=p v(x)$. This gives $-m=p v(x)$, contradicting
$p\nmid m$.
\end{proof}

\begin{lemma}[Irreducibility of an Artin--Schreier polynomial]
\label{lem:artin-schreier-irreducible}
\source{camina1997:page-0003}
\lean{Polynomial.irreducible_X_pow_sub_X_sub_C}
\leanok
Let $F$ be any field of characteristic $p$ and $a\in F$.
If $a\notin\wp(F)$, then $Y^p-Y-a$ is irreducible in $F[Y]$.
\end{lemma}

\begin{proof}
Put $f=Y^p-Y-a$. It is monic of degree $p$.
Suppose $q$ is a monic divisor of $f$ of degree $d$ with
$0<d<p$. In a splitting field of $q$, list its roots with
multiplicity as $r_1,\ldots,r_d$ and put $s=\sum_i r_i$.
Each $r_i$ is a root of $f$, so $r_i^p-r_i=a$.
The coefficient formula gives $s=-[Y^{d-1}]q\in F$.
Additivity of Frobenius in characteristic $p$ now gives
\[
 s^p-s=\sum_i(r_i^p-r_i)=d a.
\]
The element $d\cdot1$ of $F$ is nonzero because $0<d<p$,
and satisfies $d^p=d$. Consequently
\[
 (s/d)^p-s/d=(s^p-s)/d=a,
\]
contradicting $a\notin\wp(F)$.
Thus no proper monic divisor has positive degree. Every factorization
of a monic polynomial over a field can be rescaled to monic factors;
if neither factor is a unit, each has positive degree strictly less
than $p$, contrary to what was just proved. Hence $f$ is irreducible.
\end{proof}

\begin{lemma}[The cyclic extension defined by a reduced pole]
\label{lem:artin-schreier-cyclic-extension}
\uses{lem:artin-schreier-no-root-from-pole,lem:artin-schreier-irreducible}
\source{camina1997:page-0003,camina1997:page-0005}
\lean{LaurentSeries.isField_adjoinRoot_of_orderTop, Camina1997.ArtinSchreier.rootBasis, Camina1997.ArtinSchreier.sum_rootBasis_repr, Camina1997.ArtinSchreier.finrank_adjoinRoot, Camina1997.ArtinSchreier.translationAction, Camina1997.ArtinSchreier.translation_root, LaurentSeries.automorphismEquivOfPole, LaurentSeries.isGalois_adjoinRoot_of_orderTop}
\leanok
Suppose $v(\gamma)=-m<0$ and $p\nmid m$. The quotient
\[
 L=K[Y]/(Y^p-Y-\gamma)
\]
is a field of degree $p$ over $K$. Writing $y$ for the class of
$Y$, every element of $L$ has a unique expression
$\sum_{i=0}^{p-1}a_i y^i$ with $a_i\in K$.
The maps $\sigma_c(y)=y+c$, fixing $K$, identify
$(\mathbf F_p,+)$ with $\operatorname{Gal}(L/K)$.
\end{lemma}

\begin{proof}
The first two lemmas give irreducibility. Division by the monic
polynomial gives a unique remainder of degree less than $p$,
and hence the displayed basis and degree. The quotient is a
field: for a nonzero remainder $h$, irreducibility gives
$\gcd(h,Y^p-Y-\gamma)=1$, and a Bezout identity gives an inverse
to its class.
Substitution $y\mapsto y+c$ respects the defining relation,
because $c^p=c$, and substitution by $-c$ is its inverse.
These automorphisms satisfy
$\sigma_c\sigma_d=\sigma_{c+d}$ and are distinct by their
values on $y$, giving an injective homomorphism from the additive
prime field to the automorphism group. There are at most $[L:K]=p$
$K$-algebra maps from $L$ to itself: each is determined by the
image of $y$, which must be a root of its degree-$p$ minimal
polynomial. Thus the $p$ translations exhaust the automorphism
group and give the claimed group isomorphism.
The equality between the number of automorphisms and the degree
also implies that the extension is Galois. Here is that criterion
in this simple-extension setting: the images of $y$ under the $p$
distinct automorphisms are distinct and exhaust the roots of its
minimal polynomial. That polynomial therefore splits with distinct
roots in $L$. Since $L=K(y)$, it is a splitting field of a separable
polynomial and hence is normal and separable.
\end{proof}

\begin{lemma}[The valuation on the cyclic extension]
\label{lem:artin-schreier-weighted-valuation}
\uses{lem:artin-schreier-cyclic-extension}
\source{camina1997:page-0005,camina1997:page-0006}
\lean{Camina1997.ArtinSchreier.weightedValuation, Camina1997.ArtinSchreier.weightedValuation_apply, Camina1997.ArtinSchreier.weightedOrder_mul, Camina1997.ArtinSchreier.weightedOrder_algebraMap, Camina1997.ArtinSchreier.weightedOrder_root, Camina1997.ArtinSchreier.weightedValuation_surjective}
\leanok
For $L$, $y$, $m$ as above, the formula
\[
 w\left(\sum_{i=0}^{p-1}a_i y^i\right)
   =\min_{a_i\ne0}\{p v(a_i)-mi\},\qquad w(0)=+\infty,
\]
defines a surjective additive discrete valuation on $L$.
It satisfies $w|_K=p v$ and $w(y)=-m$.
\end{lemma}

\begin{proof}
The formula is well-defined by uniqueness of the polynomial
expression. Distinct nonzero terms have distinct weights: an
equality $p v(a_i)-mi=p v(a_j)-mj$ implies $p\mid m(i-j)$,
hence $p\mid i-j$ and then $i=j$, since $0\le i,j<p$.
In particular every nonzero expression has a unique least-weight
term. The inequalities for $v$ on each coefficient give
$w(x+x')\ge\min\{w(x),w(x')\}$, and clearly $w(1)=0$.

We prove multiplicativity. Let the least-weight terms of nonzero
$x=\sum a_i y^i$ and $z=\sum b_j y^j$ have indices $i_0,j_0$.
Among the nonzero terms $a_i b_j y^{i+j}$ in the unreduced
product, exactly $(i_0,j_0)$ has least weight
$w(x)+w(z)$. If $i+j<p$, this term is already in basis form.
If $p\le i+j\le2p-2$, the relation $y^p=y+\gamma$ gives
\[
 a_i b_j y^{i+j}
   =a_i b_j y^{i+j-p+1}
      +a_i b_j\gamma y^{i+j-p}.
\]
Both exponents on the right lie between $0$ and $p-1$.
The second term has the original weight
$p v(a_i)+p v(b_j)-m(i+j)$, because $v(\gamma)=-m$;
the first has that weight plus $m(p-1)>0$.
Thus reduction leaves exactly one contribution of globally least
weight, namely the lower-weight contribution of $(i_0,j_0)$.
Contributions to the same basis coefficient with larger weight
have larger coefficient valuation, so they cannot cancel its
leading coefficient. Consequently
$w(xz)=w(x)+w(z)$. The assertion also holds when a factor is zero.
Applying it to $xx^{-1}=1$ gives $w(x^{-1})=-w(x)$.

The formula gives $w(t)=p$ and $w(y)=-m$. For any integer $n$,
choose $0\le i<p$ with $mi\equiv-n\pmod p$; this is possible
because $m$ is invertible modulo $p$. Put $a=(n+mi)/p\in\mathbf Z$.
The single basis term $t^a y^i$ has weight $pa-mi=n$.
Every integer is therefore realized, and zero realizes $+\infty$.
The formula on constant basis expressions gives $w|_K=p v$.
\end{proof}

\begin{lemma}[Completeness and residue field]
\label{lem:artin-schreier-complete-residue}
\uses{lem:artin-schreier-weighted-valuation}
\source{camina1997:page-0005,camina1997:page-0006}
\lean{Camina1997.ArtinSchreier.weightedValued_completeSpace, Camina1997.ArtinSchreier.rootBasisUniformEquiv, Camina1997.ArtinSchreier.exists_unique_constant_sub_weightedOrder_pos, Camina1997.ArtinSchreier.residueFieldEquiv}
\leanok
The valued field $(L,w)$ is complete. Its residue field is
canonically $k$.
\end{lemma}

\begin{proof}
The formula for $w$ shows that a sequence
$x_n=\sum_{i<p}a_{n,i}y^i$ is Cauchy precisely when each of its
$p$ coefficient sequences is Cauchy in $K$: a lower bound on
$w(x_n-x_r)$ is exactly the corresponding lower bound on all
$p v(a_{n,i}-a_{r,i})-mi$.
Each coefficient sequence converges, since $k((t))$ is complete.
Indeed, for a Cauchy sequence of Laurent series the coefficients
in each bounded range eventually stabilize; their stabilized
values have a common lower support bound, obtained by comparing
the tail of the sequence with one fixed term. They therefore
define a Laurent series, and the Cauchy condition gives
convergence to it. Taking these limits for all $i<p$ and applying
the formula for $w$ proves convergence of $x_n$ in $L$.
The valuations define metrics by $d(x,z)=2^{-w(x-z)}$ and
$2^{-v(x-z)}$, with distance zero for equal arguments, so this
sequential completeness is completeness of the valuation
uniformity.

Let $x=\sum_{i<p}a_i y^i$ satisfy $w(x)\ge0$.
Each nonzero basis term has nonnegative weight. For $i>0$
that weight cannot be zero, since $p\nmid mi$; all such terms
lie in the maximal ideal $\{z:w(z)>0\}$. The constant basis
coefficient $a_0$ has $v(a_0)\ge0$. If $c\in k$ is its
constant coefficient as a power series in $t$, then
$v(a_0-c)>0$, so $w(x-c)>0$.
The class of every valuation-ring element is therefore represented
by a constant in $k$. A nonzero constant has valuation zero,
so distinct constants give distinct residue classes. Reduction
on constants is a field homomorphism, proving the claimed
residue-field identification.
\end{proof}

\begin{lemma}[Ramification index of the reduced-pole extension]
\label{lem:artin-schreier-total-ramification}
\uses{lem:artin-schreier-cyclic-extension,lem:artin-schreier-weighted-valuation,lem:artin-schreier-complete-residue}
\source{camina1997:page-0005,camina1997:page-0006}
\notready
The ramification index of $L/K$ is $p$, and $L/K$ is totally ramified.
\end{lemma}

\begin{proof}
The weighted valuation is surjective, so $w(L^\times)=\mathbf Z$.
Its restriction to $K$ is $pv$, and $v(K^\times)=\mathbf Z$,
so $w(K^\times)=p\mathbf Z$. The cosets of $p\mathbf Z$
are represented uniquely by $0,1,\ldots,p-1$; hence their index
is $p=[L:K]$. The residue-field identification shows that the
residue extension has degree one. These two assertions give
total ramification.
\end{proof}

\begin{lemma}[The complete valuation ring]
\label{lem:artin-schreier-integer-topology}
\uses{lem:artin-schreier-complete-residue,lem:artin-schreier-weighted-valuation}
\source{camina1997:page-0005}
\lean{Camina1997.ArtinSchreier.isClosed_weightedInteger, Camina1997.ArtinSchreier.weightedInteger_completeSpace, Camina1997.ArtinSchreier.weightedInteger_t2Space, Camina1997.ArtinSchreier.weightedInteger_isLinearTopology, Camina1997.ArtinSchreier.weightedInteger_isTopologicallyNilpotent}
\leanok
The valuation ring $\mathcal O=\{x\in L:w(x)\ge0\}$ is a complete
Hausdorff topological ring. Its neighborhoods of zero have a
basis of ideals. If $w(\pi)=1$, then $\pi^n\longrightarrow0$
in $\mathcal O$.
\end{lemma}

\begin{proof}
The nonarchimedean inequality and multiplicativity show that
$\mathcal O$ is a subring and that, for every integer $r$,
$\{x\in\mathcal O:w(x)>r\}$ is an ideal. These sets form a
neighborhood basis at zero for the induced topology.
If $w(x)<0$, every $z$ with $w(z-x)>w(x)$ satisfies
$w(z)=w(x)<0$. Thus the complement of $\mathcal O$ is open,
and $\mathcal O$ is closed in the complete field $L$.
A Cauchy filter on this closed subspace converges in $L$ and
its limit remains in the subspace, proving completeness.
The valuation separates distinct points, so the topology
is Hausdorff. Finally $w(\pi^n)=n$, which eventually exceeds
every prescribed integer bound.
\end{proof}

\begin{lemma}[A Laurent-series parameter for the extension]
\label{lem:artin-schreier-laurent-parameter}
\uses{lem:artin-schreier-complete-residue,lem:artin-schreier-integer-topology}
\source{camina1997:page-0005}
\lean{Camina1997.ArtinSchreier.powerSeriesEval, Camina1997.ArtinSchreier.weightedOrder_powerSeriesEval, Camina1997.ArtinSchreier.powerSeriesEval_uniformizerSeries, Camina1997.ArtinSchreier.powerSeriesEquiv, Camina1997.ArtinSchreier.laurentParameterEquiv, Camina1997.ArtinSchreier.weightedOrder_laurentParameterEquiv, Camina1997.ArtinSchreier.laurentParameterHomeomorph}
\leanok
For every $\pi\in L$ with $w(\pi)=1$, the map
\[
 k((T))\longrightarrow L,\qquad
 \sum_{n\ge n_0}c_n T^n\longmapsto\sum_{n\ge n_0}c_n\pi^n
\]
is an isomorphism of topological $k$-algebras. It identifies
the order in $T$ with $w$.
\end{lemma}

\begin{proof}
First evaluate power series into the valuation ring. Completeness,
the linear valuation topology, and $w(\pi^n)=n$ give the
convergent evaluation homomorphism $k[[T]]\to\mathcal O_L$.
It fixes constants and sends $T$ to $\pi$.
For a nonzero power series write $f=T^r u$, where
$r$ is its order and $u$ is a unit. A ring homomorphism sends
units to units, and every unit of $\mathcal O_L$ has value
zero: both it and its inverse have nonnegative value, whose
sum is zero. Hence evaluation sends $f$ to an element of
value $r$. In particular this evaluation is injective.

To prove surjectivity, first take $x$ with $w(x)\ge0$.
Put $x_0=x$. The residue-field assertion supplies a unique
$c_n\in k$ with $w(x_n-c_n)>0$; set
$x_{n+1}=(x_n-c_n)/\pi$. Then $w(x_{n+1})\ge0$ and induction gives
\[
 x=\sum_{j=0}^{N-1}c_j\pi^j+\pi^N x_N.
\]
The remainder has valuation at least $N$, so the partial sums
converge to $x$. Thus evaluation is an isomorphism $k[[T]]\simeq\mathcal O_L$.
The fraction fields of these rings are respectively $k((T))$
and $L$, so localization extends it to a $k$-algebra isomorphism
between these fields. For a quotient $a/b$ of power series,
additivity of both valuations gives preservation of order from
the power-series case. This also identifies the extension with
the displayed Laurent evaluation. Finally preservation of order
makes both this map and its inverse uniformly continuous for
the valuation topologies.
\end{proof}

\begin{lemma}[Depth of an Artin--Schreier translation]
\label{lem:artin-schreier-translation-depth}
\uses{lem:artin-schreier-cyclic-extension,lem:artin-schreier-weighted-valuation,lem:artin-schreier-complete-residue}
\source{camina1997:page-0008}
\lean{Camina1997.ArtinSchreier.weightedOrder_translation, Camina1997.ArtinSchreier.uniformContinuous_translation, Camina1997.ArtinSchreier.weightedOrder_rootMonomial_eq_one, Camina1997.ArtinSchreier.weightedOrder_translation_sub_uniformizer, Camina1997.ArtinSchreier.exists_uniformizer_translation_depth}
\leanok
Choose $1\le b<p$ and an integer $a$ with $ap-bm=1$, and put
$\pi=t^a y^b$. Then $w(\pi)=1$, every $\sigma_c$ is a
valuation-preserving continuous automorphism of $L$, and for
$c\in\mathbf F_p^\times$,
\[
 w(\sigma_c(\pi)-\pi)=m+1.
\]
\end{lemma}

\begin{proof}
Multiplication by $m$ permutes the nonzero residue classes modulo
$p$, so $b$ exists with $bm\equiv-1\pmod p$; taking
$a=(bm+1)/p$ gives the desired identity and $w(\pi)=1$.
Since $w(y)=-m<0$ and every constant has nonnegative valuation,
$w(y+c)=w(y)$. Multiplicativity then gives
$w(a_i(y+c)^i)=w(a_i y^i)$ for each basis term. The inequality
for finite sums implies $w(\sigma_c(x))\ge w(x)$.
Apply the same inequality to $\sigma_{-c}$ and $\sigma_c(x)$
to obtain equality. Applying equality to differences proves
uniform continuity of $\sigma_c$, including continuity of its
inverse $\sigma_{-c}$.

For $c\ne0$, expand the displacement directly:
\[
 \sigma_c(\pi)-\pi
 =t^a\sum_{j=0}^{b-1}\binom bj y^j c^{b-j}.
\]
The term of degree $b-1$ has nonzero coefficient $bc$,
because $1\le b<p$. Its weighted order before multiplication
by $t^a$ is $-m(b-1)$. Every other nonzero term has degree
$j<b-1$, and hence strictly larger weighted order $-mj$.
The unique least term cannot cancel. The displacement therefore
has order $ap-m(b-1)=1+m$, as claimed.
\end{proof}

\begin{theorem}[Cyclic subgroups with prescribed prime-to-$p$ depth]
\label{thm:nottingham-cyclic-prescribed-depth}
\uses{lem:artin-schreier-translation-depth,lem:artin-schreier-laurent-parameter,lem:nottingham-laurent-equivalence,def:nottingham-depth,lem:nottingham-depth-filtration}
\source{camina1997:page-0005,camina1997:page-0008}
\notready
For every positive $m$ with $p\nmid m$, the Nottingham group
$J_p$ contains a cyclic subgroup of order $p$ whose nonidentity
elements all have depth $m$.
\end{theorem}

\begin{proof}
Take $\gamma=t^{-m}$ and the field and parameter constructed
above. Transport the automorphisms $\sigma_c$ to $k((T))$
through the parameter isomorphism. They are continuous
$k$-algebra automorphisms, and their parameter differences have
order $m+1\ge2$ when $c\ne0$; for $c=0$ the difference is zero.
A Laurent series of order at least two is $T^2$ times a power
series. The Laurent--Nottingham equivalence therefore identifies
each transported automorphism with an element of $J_p$ and
respects their composition. The translations are distinct and
compose according to addition in $\mathbf F_p$, so their image
is cyclic of order $p$. The order formula for every nonzero $c$
gives exactly depth $m$.
\end{proof}
