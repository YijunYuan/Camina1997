\section{Compactness of continuous homomorphisms}

The compactness step in \cite{Camina1997}, Theorem~5, can be expressed
entirely in finite quotients. The argument below replaces the paper's
evaluation of infinite pro-$p$ words and its appeal to automatic
continuity by a diagonal subsequence of continuous quotient maps.
The compactness criterion keeps depth control on the finite embeddings
as an explicit hypothesis.

\begin{lemma}[Finite targets and finitely many generators]
\label{lem:finite-continuous-hom-set}
\uses{def:topologically-finitely-generated}
\source{camina1997:page-0011}
\lean{Camina1997.continuousMonoidHom_ext_of_dense_closure, Camina1997.continuousMonoidHom_eval_injective, Camina1997.card_continuousMonoidHom_le_of_dense_tuple, Camina1997.TopologicallyFG.finite_continuousMonoidHom}
\leanok
Let $P$ be a topological group with a finite tuple
$(s_1,\ldots,s_r)$ generating a dense subgroup. If $A$ is a finite
discrete group, there are at most $|A|^r$ continuous homomorphisms
$P\to A$.
\end{lemma}

\begin{proof}
Two homomorphisms with the same values on the tuple agree on the
subgroup it generates, by preservation of products and inverses.
If both are continuous, their equalizer is closed: it is the
inverse image of the diagonal under their product map, and the
diagonal in $A\times A$ is closed because $A$ is Hausdorff.
The equalizer contains a dense subgroup and hence all of $P$.
Thus evaluation at the tuple injects the continuous homomorphism
set into $A^r$, which has $|A|^r$ elements.
\end{proof}

\begin{lemma}[Countability of the finite quotient basis]
\label{lem:finitely-generated-pro-p-countable-base}
\uses{def:pro-p-group,def:countably-based,lem:finite-continuous-hom-set}
\source{camina1997:page-0010,camina1997:page-0011}
\lean{Camina1997.TopologicallyFG.countable_openNormalSubgroup, Camina1997.TopologicallyFG.isCountablyGenerated_nhds_one}
\leanok
A compact topologically finitely generated group has only countably
many open normal subgroups. In particular, every topologically
finitely generated pro-$p$ group is countably based.
\end{lemma}

\begin{proof}
Let $P$ be compact and topologically finitely generated.
For each natural number $d$, let $S_d$ be the finite symmetric
group on $\{1,\ldots,d\}$, given the discrete topology.
There are only finitely many continuous homomorphisms $P\to S_d$
by Lemma~\ref{lem:finite-continuous-hom-set}. Their union over
all $d$ is therefore countable. Each such homomorphism has an
open normal kernel: normality is algebraic, and openness follows
by taking the inverse image of the open singleton $\{1\}$.

Conversely, let $N$ be an open normal subgroup of $P$.
The open cosets of $N$ cover $P$, so compactness gives finitely
many cosets. Thus $Q=P/N$ is finite, and its quotient topology
is discrete because each coset is open. The left regular action
$Q\to\operatorname{Sym}(Q)$ is injective: if an element acts
trivially, its action on the identity gives that it is the identity.
Label the elements of $Q$ by $\{1,\ldots,d\}$, where $d=|Q|$.
Composing the quotient projection with this faithful permutation
action gives a continuous homomorphism $P\to S_d$ whose kernel
is exactly $N$. Consequently the preceding countable family of
kernels contains every open normal subgroup of $P$.

If $P$ is pro-$p$, its defining identity neighbourhood basis
consists of open normal subgroups. Since there are only countably
many such subgroups, this is a countable basis. This finite
permutation argument supplies the countability needed for the
tower and subsequence constructions in Camina's Theorem~5.
\end{proof}

\begin{lemma}[A separating countable tower of finite quotients]
\label{lem:pro-p-countable-quotient-tower}
\uses{def:pro-p-group,def:countably-based}
\source{camina1997:page-0010}
\lean{Camina1997.IsProPGroup.exists_antitone_openNormalBasis, Camina1997.IsProPGroup.antitone_of_openNormalBasis, Camina1997.IsProPGroup.exists_le_of_openNormalBasis, Camina1997.IsProPGroup.iInf_eq_bot_of_openNormalBasis, Camina1997.IsProPGroup.eventually_not_mem_openNormalBasis, Camina1997.IsProPGroup.finite_quotient, Camina1997.IsProPGroup.isPGroup_quotient, OpenNormalSubgroup.quotientProjection, OpenNormalSubgroup.quotientProjection_surjective, OpenNormalSubgroup.quotientTransition, OpenNormalSubgroup.quotientTransition_surjective, OpenNormalSubgroup.quotientTransition_comp_projection, OpenNormalSubgroup.quotientTransition_comp, OpenNormalSubgroup.quotientTransition_self}
\leanok
A countably based pro-$p$ group $P$ has a decreasing sequence
$(N_n)_{n\ge0}$ of open normal subgroups of finite $p$-power index
forming an identity neighbourhood basis. This sequence is cofinal
among open normal subgroups, and
\[
 \bigcap_{n\ge0}N_n=1.
\]
For each $x\ne1$ there is $n$ such that $x\notin N_m$ for every
$m\ge n$. Each quotient $P_n=P/N_n$ is a finite discrete $p$-group.
The quotient projections $\pi_n:P\to P_n$ and the transition maps
\[
 q_{m,n}:P_m\longrightarrow P_n,\qquad xN_m\longmapsto xN_n
 \quad(m\ge n),
\]
are continuous surjective homomorphisms, with
$q_{m,n}\pi_m=\pi_n$, $q_{n,n}=\operatorname{id}$, and
$q_{m,n}q_{\ell,m}=q_{\ell,n}$ for $\ell\ge m\ge n$.
\end{lemma}

\begin{proof}
Choose a countable identity neighbourhood basis $(U_n)$.
The pro-$p$ basis property allows a choice of an open normal
subgroup $N_0\subseteq U_0$ of $p$-power index. Recursively choose
$N_{n+1}$ of the same kind inside $N_n\cap U_{n+1}$.
Then the sequence is decreasing and is a basis, because
$N_n\subseteq U_n$. Every open normal subgroup is an identity
neighbourhood, so it contains some $N_n$.

Since $P$ is Hausdorff, for each $x\ne1$ there is an identity
neighbourhood omitting $x$. One of the $N_n$ is contained in
that neighbourhood, and all $N_m$ for $m\ge n$ are contained
in $N_n$. This proves both separation assertions.
The quotient $P_n$ has order $[P:N_n]$, a power of $p$,
and so is a finite $p$-group. Its cosets are open, which gives
the discrete quotient topology and continuity of $\pi_n$.

For $m\ge n$, the inclusion $N_m\subseteq N_n$ makes
$q_{m,n}$ well-defined. It preserves multiplication, and every
coset $xN_n$ is the image of $xN_m$, proving surjectivity.
It is continuous because $P_m$ is discrete. The same coset
calculation proves all three displayed compatibility identities;
the projections themselves are surjective by their definition.
This supplies the separating finite-quotient tower used at the
start of Camina's Theorem~5, without asserting a further
inverse-limit identification of $P$.
\end{proof}

\begin{lemma}[A diagonal limit of continuous Nottingham homomorphisms]
\label{lem:nottingham-hom-diagonal-limit}
\uses{lem:finite-continuous-hom-set,lem:nottingham-quotient-card,lem:nottingham-inverse-limit}
\source{camina1997:page-0011,camina1997:page-0012}
\lean{Camina1997.Nottingham.exists_continuousMonoidHom_of_compatible, Camina1997.Nottingham.exists_diagonal_limit}
\leanok
Let $P$ be a topologically finitely generated topological group and
let $f_i:P\to J_p$, for $i\in\mathbf N$, be continuous homomorphisms.
There are a strictly increasing sequence $(i_j)$ and a continuous
homomorphism $f:P\to J_p$ such that, for every $n\ge0$, the maps
\[
 P\xrightarrow{f_{i_j}}J_p\longrightarrow J_p/J_{p,n}
\]
are equal to the quotient of $f$ for all sufficiently large $j$.
The threshold may be chosen independently of the element of $P$.
\end{lemma}

\begin{proof}
Write $Q_n=J_p/J_{p,n}$ and let $\pi_n:J_p\to Q_n$ be the
continuous quotient map. The set of continuous homomorphisms
$P\to Q_n$, denoted $H_n$, is finite by the first lemma and
finiteness of $Q_n$. Give each $H_n$ the discrete topology.
The countable product $\prod_n H_n$ is compact and metrizable:
compactness follows from compactness of each finite factor, and
a compatible metric is
\[
 d(a,b)=\sum_{n\ge0}2^{-n-1}\mathbf 1_{a_n\ne b_n}.
\]
It is therefore sequentially compact. Apply this to the sequence
$u_i=(\pi_nf_i)_n$. There are a strictly increasing sequence
$(i_j)$ and a point $h=(h_n)_n$ such that $u_{i_j}\to h$.
Each coordinate projection is continuous, so
$\pi_nf_{i_j}\to h_n$ in the discrete space $H_n$.
The open singleton $\{h_n\}$ then gives an index $j_n$ such that
$\pi_nf_{i_j}=h_n$ as whole homomorphisms for every $j\ge j_n$.

The maps $h_n$ are compatible. Indeed, for $m\ge n$, choose
$j\ge\max(j_m,j_n)$. Both $h_m$ and $h_n$ are quotients of the same map
$f_{i_j}$, so $\pi_{m,n}h_m=h_n$, where $\pi_{m,n}:Q_m\to Q_n$
is the natural quotient homomorphism.
For each $x\in P$, the tuple $(h_n(x))_n$ therefore lies in
the congruence inverse limit. It depends homomorphically on $x$,
since every coordinate does. It also depends continuously on $x$:
all coordinates are continuous, and product continuity followed
by restriction to the inverse-limit subspace gives continuity.
Compose this map with the inverse of the topological group
isomorphism $J_p\cong\varprojlim Q_n$. This gives the required
continuous homomorphism $f$. By construction $\pi_nf=h_n$, and
the equality for all $j\ge j_n$ gives the claimed uniform threshold.
\end{proof}

\begin{lemma}[Depth control prevents collapse in the limit]
\label{lem:nottingham-depth-controlled-limit-embedding}
\uses{lem:nottingham-hom-diagonal-limit,lem:compact-injection-closed-embedding}
\source{camina1997:page-0012}
\lean{Camina1997.Nottingham.exists_closedEmbedding_of_eventually_not_mem}
\leanok
Let $P$ be a compact, topologically finitely generated topological
group. Suppose that $(f_i:P\to J_p)_{i\in\mathbf N}$ is a sequence
of continuous homomorphisms such that for every $x\ne1$ there
exist $n,N\in\mathbf N$ with
\[
 f_i(x)\notin J_{p,n}\qquad\text{for every }i\ge N.
\]
Then $P$ admits a closed group embedding in $J_p$.
\end{lemma}

\begin{proof}
Take $f$ and the strictly increasing subsequence $(i_j)$ from
the preceding lemma. Suppose $x\ne1$, and choose $n,N$ as in
the hypothesis. For all sufficiently large $j$, we have
$i_j\ge N$ and $\pi_nf_{i_j}=\pi_nf$. The first assertion holds
because a strictly increasing sequence of natural numbers is
unbounded; the second is the diagonal-limit conclusion.
The hypothesis gives $\pi_nf_{i_j}(x)\ne1$, and hence
$\pi_nf(x)\ne1$. Thus $f(x)\ne1$. Its kernel is trivial, so
$f$ is injective. It is continuous by its construction, $P$ is
compact by hypothesis, and $J_p$ is Hausdorff. The compact-source
lemma therefore makes $f$ a homeomorphism onto a closed subgroup.
\end{proof}

\begin{lemma}[Depth-preserving quotient embeddings give a closed embedding]
\label{lem:nottingham-embedding-of-quotient-depths}
\uses{lem:pro-p-countable-quotient-tower,lem:nottingham-depth-filtration,lem:nottingham-depth-controlled-limit-embedding}
\source{camina1997:page-0010,camina1997:page-0011,camina1997:page-0012}
\lean{Camina1997.Nottingham.exists_closedEmbedding_of_quotient_depths}
\leanok
Let $P$ be a topologically finitely generated pro-$p$ group, and
let $(N_n)$ be a decreasing identity neighbourhood basis of open
normal subgroups. Write $P_n=P/N_n$, let $\pi_n:P\to P_n$ be
the projections, and let $q_n:P_{n+1}\to P_n$ be the transition maps.
Suppose there are injective homomorphisms $\varphi_n:P_n\to J_p$
such that for every $n$ and $z\in P_{n+1}$,
\[
 q_n(z)\ne1\quad\Longrightarrow\quad
 D(\varphi_{n+1}(z))=D(\varphi_n(q_n(z))).
\]
Then $P$ admits a continuous injective homomorphism into $J_p$
which is a homeomorphism onto a closed subgroup.
\end{lemma}

\begin{proof}
Each $P_n$ is discrete, so $f_n=\varphi_n\pi_n$ is a continuous
homomorphism. Fix $x\ne1$. The decreasing basis separates points,
so there is $n$ with $\pi_m(x)\ne1$ for all $m\ge n$.
Injectivity of $\varphi_n$ gives $f_n(x)\ne1$; write its finite
depth as $D(f_n(x))=d$.
For $m\ge n$, the compatibility of quotient projections gives
$q_m(\pi_{m+1}(x))=\pi_m(x)\ne1$. The assumed equality of depths
therefore gives
\[
 D(f_{m+1}(x))=D(f_m(x)).
\]
Induction on $m\ge n$ proves that $D(f_m(x))=d$ for every
$m\ge n$. Membership in $J_{p,d+1}$ is equivalent to having
depth at least $d+1$, so $f_m(x)\notin J_{p,d+1}$ for all such $m$.
The avoidance criterion of
Lemma~\ref{lem:nottingham-depth-controlled-limit-embedding}
now gives the required closed embedding.

This is the finite-quotient form of the noncollapse argument in
Camina's Theorem~5. Its hypothesis asks for successive depth
equalities; the maps $\varphi_n$ need not agree coefficientwise.
\end{proof}

\begin{lemma}[Simultaneous depth control along a finite-group tower]
\label{lem:witt-depth-compatible-finite-group-sequence}
\uses{thm:witt-finite-laurent-realization,lem:laurent-action-nottingham-restriction,thm:witt-depth-preserving-finite-quotient-lift}
\source{camina1997:page-0010,camina1997:page-0011}
\lean{Camina1997.Witt.exists_depth_compatible_sequence}
\leanok
Let $(G_n)_{n\ge0}$ be a sequence of finite $p$-groups with
surjective homomorphisms $q_n:G_{n+1}\to G_n$.
There is a sequence of injective homomorphisms
$\varphi_n:G_n\to J_p$ such that, simultaneously for all $n$
and $g\in G_{n+1}$,
\[
 q_n(g)\ne1\quad\Longrightarrow\quad
 D(\varphi_{n+1}(g))=D(\varphi_n(q_n(g))).
\]
\end{lemma}

\begin{proof}
Put $K=\mathbf F_p((t))$ and use $F=K$ as a fixed abstract base
field, with its identity isomorphism to the Laurent field.
At stage $n$ we retain a field embedding $\alpha_n:F\hookrightarrow K$
making $K/F$ finite Galois, an isomorphism
\[
 \rho_n:G_n\simeq\operatorname{Gal}_{\alpha_n}(K/F),
\]
and an injective homomorphism $\varphi_n:G_n\to J_p$
such that $\widetilde{\varphi_n(g)}=\rho_n(g)$ on $K$.

The finite Laurent realization theorem constructs $\alpha_0$
and $\rho_0$ with an order-preserving action. Restricting this
action to power series and normalizing the $p$-group action
by Lemma~\ref{lem:laurent-action-nottingham-restriction}
gives the required $\varphi_0$, including its injectivity
and agreement with $\rho_0$ on the Laurent field.

Suppose the data at stage $n$ have been constructed.
Apply Theorem~\ref{thm:witt-depth-preserving-finite-quotient-lift}
to this full realization, its Nottingham action $\varphi_n$,
and the surjection $q_n$. It constructs $\alpha_{n+1}$,
$\rho_{n+1}$, and $\varphi_{n+1}$ with all the retained
properties. It also gives the required depth equality whenever
$\varphi_n(q_n(g))\ne1$. Injectivity of $\varphi_n$ identifies
this condition with $q_n(g)\ne1$.

Choose one such successor for each possible stage realization
and apply recursion on $n$. The resulting sequence has the
retained properties at every stage and the depth equality at
every transition. Keeping the full Galois realization at each
stage supplies the hypotheses for the next application of the
lifting theorem. This makes explicit the successive Witt
constructions in Camina's proof of Theorem~5.
\end{proof}

\begin{theorem}[Finitely generated pro-$p$ groups embed in Nottingham]
\label{thm:nottingham-finitely-generated-pro-p}
\uses{lem:finitely-generated-pro-p-countable-base,lem:pro-p-countable-quotient-tower,lem:witt-depth-compatible-finite-group-sequence,lem:nottingham-embedding-of-quotient-depths}
\source{camina1997:page-0010,camina1997:page-0011,camina1997:page-0012}
\lean{Camina1997.Nottingham.exists_closedEmbedding_of_topologicallyFG}
\leanok
For every prime $p$ and every topologically finitely generated
pro-$p$ group $P$, there is a continuous injective homomorphism
$P\to J_p$ that is a homeomorphism onto a closed subgroup.
\end{theorem}

\begin{proof}
The countability lemma gives a countable identity neighbourhood
basis. Choose a decreasing basis $(N_n)$ of open normal subgroups
and the associated surjective quotient maps
$q_n:P/N_{n+1}\to P/N_n$ by
Lemma~\ref{lem:pro-p-countable-quotient-tower}.
Each quotient is a finite $p$-group. Apply
Lemma~\ref{lem:witt-depth-compatible-finite-group-sequence}
to obtain injective maps $\varphi_n:P/N_n\to J_p$ preserving
depth across every transition at every element with nonidentity
image. These are exactly the hypotheses of
Lemma~\ref{lem:nottingham-embedding-of-quotient-depths}, which
therefore gives the required continuous closed embedding.
\end{proof}
