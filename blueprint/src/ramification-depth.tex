\section{Displacement orders and preservation of depth}

Write $v$ for the order on $k[[X]]$, with $v(0)=+\infty$.
A normalized $k$-algebra automorphism $g$ sends
$X$ to $X+\delta$, where $v(\delta)\ge2$.
The following coefficient arguments supply the cancellation details in
\cite{Camina1997}, Lemmas~4--5. Automorphisms act on the left here;
this changes no displacement orders from the paper's right-action convention.

\begin{lemma}[Displacement of a monomial]
\label{lem:normalized-monomial-displacement}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.order_pow_sub_pow_of_linear_coeff_one}
\leanok
Let $k$ be a field, let $y\in k[[X]]$ have constant coefficient zero
and coefficient of $X$ equal to one, and suppose
$v(y-X)=n+1<\infty$. If $m$ is a natural number whose image in $k$
is nonzero, then
\[
 v(y^m-X^m)=m+n.
\]
\end{lemma}

\begin{proof}
The hypothesis on $m$ implies $m\ge1$. Write $y=Xu$, where
$u(0)=1$. The finite geometric identity gives
\[
 y^m-X^m=X^{m-1}(y-X)(1+u+\cdots+u^{m-1}).
\]
The last factor has constant coefficient $m\ne0$, hence has order zero.
The order of a product of nonzero power series is the sum of their orders.
Consequently the displayed product has order
$(m-1)+(n+1)+0=m+n$.
\end{proof}

\begin{lemma}[Displacement at a leading exponent]
\label{lem:normalized-leading-displacement}
\uses{lem:normalized-monomial-displacement}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.order_algEquiv_sub_of_order}
\leanok
Let $g$ be a normalized automorphism of $k[[X]]$ with
$v(g(X)-X)=n+1<\infty$. If $v(f)=m<\infty$ and the image
of $m$ in $k$ is nonzero, then
\[
 v(g(f)-f)=m+n.
\]
\end{lemma}

\begin{proof}
Write $f=X^m u$, where $u(0)\ne0$. Every power-series automorphism
preserves constant coefficients, so $g(u)$ has order zero.
Congruence on $X$ modulo $X^{n+1}$ implies the same congruence on
all power series. To see this, it holds for each monomial by
factoring $g(X)^j-X^j$ by $g(X)-X$, and therefore for polynomials;
for a power series, truncate after degree $n$, since both the original
series and its image differ from these polynomials by multiples of
$X^{n+1}$. Thus $v(g(u)-u)\ge n+1$.
Now
\[
 g(f)-f=(g(X)^m-X^m)g(u)+X^m(g(u)-u).
\]
The first summand has order $m+n$ by the preceding lemma, while the
second has order at least $m+n+1$. Their leading terms cannot cancel,
which proves the assertion.
\end{proof}

\begin{lemma}[Displacement of a Frobenius split]
\label{lem:frobenius-split-displacement}
\uses{lem:normalized-leading-displacement}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.displacement_orders_of_frobenius_split, PowerSeries.le_order_displacement_of_frobenius_split, PowerSeries.order_displacement_of_frobenius_split}
\leanok
Let $p$ be prime and $k$ a field of characteristic $p$. Let $g$ be a normalized
$k$-algebra automorphism of $k[[X]]$, and suppose
$v(g(X)-X)=n+1<\infty$. Let $A,B\in k[[X]]$ satisfy
$v(A)=1$ and $v(B)=u<\infty$, where the image of $u$ in $k$
is nonzero. For $T=A^p+B$,
\[
 v\bigl((g(A)-A)^p\bigr)=p(n+1),\qquad
 v(g(B)-B)=u+n,
\]
and
\[
 v(g(T)-T)\ge\min\{p(n+1),u+n\}.
\]
If $p(n+1)\ne u+n$, the last inequality is an equality.
\end{lemma}

\begin{proof}
The leading-displacement lemma applied first to $A$ and then to
$B$ gives $v(g(A)-A)=n+1$ and $v(g(B)-B)=u+n$.
Multiplicativity of order gives the stated order of the $p$th
power. As $g$ respects addition and powers, and Frobenius is
additive in characteristic $p$,
\[
 g(T)-T=(g(A)-A)^p+(g(B)-B).
\]
The order of a sum is at least the minimum of the two orders.
When these orders differ, the unique term of lower order has
a nonzero leading coefficient which the other summand cannot
cancel. The sum then has precisely that lower order.
\end{proof}

\begin{lemma}[A fixed Frobenius split determines its leading exponent]
\label{lem:frobenius-split-fixed-break}
\uses{lem:frobenius-split-displacement}
\source{camina1997:page-0006}
\lean{PowerSeries.first_exponent_eq_of_fixed_frobenius_split}
\leanok
Under the hypotheses of Lemma~\ref{lem:frobenius-split-displacement},
if $g(A^p+B)=A^p+B$, then $u=n(p-1)+p$.
\end{lemma}

\begin{proof}
If $p(n+1)\ne u+n$, the exact-order conclusion would make
$g(A^p+B)-(A^p+B)$ have finite order, contrary to its being zero.
Hence $p(n+1)=u+n$, and subtracting $n$ gives the asserted formula.
\end{proof}

\begin{lemma}[Depth equality for a Frobenius split]
\label{lem:frobenius-split-depth-equality}
\uses{lem:frobenius-split-displacement}
\source{camina1997:page-0007}
\lean{PowerSeries.depth_eq_of_frobenius_split}
\leanok
Under the hypotheses of Lemma~\ref{lem:frobenius-split-displacement},
suppose there are natural numbers $r,d$ such that
\[
 u=r(p-1)+p,\qquad d<r,\qquad
 v(g(A^p+B)-(A^p+B))=p(d+1).
\]
Then $n=d$.
\end{lemma}

\begin{proof}
If $n\ge r$, then
\[
 p(n+1)\ge p(r+1)>p(d+1),\qquad
 u+n=r(p-1)+p+n\ge p(r+1)>p(d+1).
\]
The lower bound of Lemma~\ref{lem:frobenius-split-displacement}
would contradict the prescribed displacement order. Thus $n<r$.
Since $p>1$, we obtain
\[
 p(n+1)<r(p-1)+p+n=u+n.
\]
The exact-order conclusion of that lemma now gives
$p(d+1)=p(n+1)$. Cancelling the positive factor $p$ gives $d=n$.
\end{proof}

\begin{lemma}[The Frobenius decomposition of a degree-$p$ parameter]
\label{lem:degree-p-parameter-frobenius-decomposition}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.eq_frobeniusRootPart_pow_add, PowerSeries.order_frobeniusRootPart_of_order_eq_prime, PowerSeries.order_frobeniusRemainder, PowerSeries.exists_frobeniusRemainder_order, PowerSeries.exists_frobeniusRemainder_order_gt_prime}
\leanok
Let $k=\mathbf F_p$ and let
\[
 T=\sum_{i\ge p}a_iX^i,\qquad a_p\ne0.
\]
Suppose $u$ is the least index not divisible by $p$ for which
$a_u\ne0$. Then there are $A,B\in k[[X]]$ such that
\[
 T=A^p+B,\qquad v(A)=1,\qquad v(B)=u,
\]
and the image of $u$ in $k$ is nonzero. If $T$ is not a $p$th
power in $k[[X]]$, such an index $u$ exists and satisfies $u>p$.
\end{lemma}

\begin{proof}
Define
\[
 A=\sum_{j\ge1}a_{pj}X^j,\qquad
 B=\sum_{\substack{i\ge p\\p\nmid i}}a_iX^i.
\]
In characteristic $p$, the $p$th power of a power series has
coefficient $a^p$ in degree $pj$ when the original coefficient
in degree $j$ is $a$, and zero coefficients in other degrees.
Indeed the identity holds for finite truncations by additivity
of Frobenius; every fixed coefficient of the original series and
its $p$th power is determined by a sufficiently long finite
truncation. Since $a^p=a$ in $\mathbf F_p$, the coefficients
of $A^p$ are exactly those of $T$ in degrees divisible by $p$.
The complementary coefficients are precisely those of $B$,
so $T=A^p+B$. The constant term of $A$ vanishes and its
coefficient of $X$ is $a_p\ne0$, giving $v(A)=1$.
The definition of $u$ makes all coefficients of $B$ below $u$
zero and its coefficient of $X^u$ nonzero, giving $v(B)=u$.
Finally $p\nmid u$ says exactly that the image of $u$ in $k$
is nonzero.

For the final assertion, if $B=0$ the identity $T=A^p+B$ would
make $T$ a $p$th power. Thus $B\ne0$, so its nonzero coefficients
have a least index $u$. The defining coefficients of $B$ imply
$p\nmid u$ and that $a_u\ne0$ is the first nonzero coefficient
of $T$ at such an index. Since $v(T)=p$, there are no nonzero
coefficients below $p$, and $p\mid p$ excludes $u=p$.
Hence $u>p$.
\end{proof}

\begin{lemma}[The two possible leading displacement orders]
\label{lem:degree-p-parameter-displacement}
\uses{lem:degree-p-parameter-frobenius-decomposition,lem:frobenius-split-displacement}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.le_order_displacement_of_first_nondivisible, PowerSeries.order_displacement_of_first_nondivisible}
\leanok
Let $k=\mathbf F_p$ and let
\[
 T=\sum_{i\ge p}a_iX^i,\qquad a_p\ne0.
\]
Suppose $u$ is the least index not divisible by $p$ for which $a_u\ne0$.
Let $g$ be normalized with $v(g(X)-X)=n+1<\infty$.
Then
\[
 v(g(T)-T)\ge\min\{p(n+1),u+n\}.
\]
If $p(n+1)\ne u+n$, equality holds.
\end{lemma}

\begin{proof}
Lemma~\ref{lem:degree-p-parameter-frobenius-decomposition}
gives $T=A^p+B$ with $v(A)=1$, $v(B)=u$, and $u$ nonzero
in $k$. Apply Lemma~\ref{lem:frobenius-split-displacement}
to obtain both assertions.
\end{proof}

\begin{lemma}[The first exponent detects the degree-p ramification break]
\label{lem:camina-four-depth-equation}
\uses{lem:degree-p-parameter-frobenius-decomposition,lem:frobenius-split-fixed-break}
\source{camina1997:page-0006}
\lean{PowerSeries.first_exponent_eq_of_fixed_parameter}
\leanok
Under the preceding hypotheses, if $g(T)=T$, then
\[
 u=n(p-1)+p.
\]
\end{lemma}

\begin{proof}
Take the decomposition $T=A^p+B$ supplied by
Lemma~\ref{lem:degree-p-parameter-frobenius-decomposition}.
Its orders satisfy all the hypotheses of
Lemma~\ref{lem:frobenius-split-fixed-break}, and $g(T)=T$ is
its fixed-point hypothesis. That lemma gives the formula.
\end{proof}

\begin{lemma}[Depth below a new ramification break is unchanged]
\label{lem:camina-five-depth-preservation}
\uses{lem:degree-p-parameter-frobenius-decomposition,lem:frobenius-split-depth-equality}
\source{camina1997:page-0007}
\lean{PowerSeries.depth_eq_of_first_nondivisible}
\leanok
Under the degree-$p$ parameter hypotheses, let $d,r$ be natural
numbers with $d<r$, and suppose
\[
 u=r(p-1)+p,\qquad v(g(T)-T)=p(d+1).
\]
If $v(g(X)-X)=n+1<\infty$, then $n=d$.
\end{lemma}

\begin{proof}
Take the decomposition $T=A^p+B$ of
Lemma~\ref{lem:degree-p-parameter-frobenius-decomposition}.
Lemma~\ref{lem:frobenius-split-depth-equality} applies with the
given displacement order and gives $n=d$.

The valuation hypothesis is the form of Camina's Lemma~5 used here.
Her displayed expansion
$g(T)-T=\sum_{j\ge d+1}\beta_jT^j$, with $\beta_{d+1}\ne0$,
has this order because it factors as $T^{d+1}V(T)$, where
$V(Z)=\beta_{d+1}+\beta_{d+2}Z+\cdots$. As $v(T)=p>0$,
substitution converges, $V(T)$ has nonzero constant coefficient,
and the product has order $p(d+1)$. In a field tower, the same
order is obtained directly by evaluating the displacement of the
lower parameter, as in the coordinate lemmas below.
\end{proof}


\begin{lemma}[A fixing automorphism detects depth through a degree-$p$ parameter]
\label{lem:nottingham-depth-fixed-parameter}
\uses{lem:normalized-leading-displacement,lem:degree-p-parameter-frobenius-decomposition,lem:camina-four-depth-equation,lem:camina-five-depth-preservation,lem:nottingham-depth-filtration}
\source{camina1997:page-0006,camina1997:page-0007}
\lean{PowerSeries.not_pow_prime_of_fixed_parameter, PowerSeries.depth_eq_of_fixed_parameter, Camina1997.Nottingham.order_sub_eq_of_depth_eq, Camina1997.Nottingham.depth_eq_of_fixed_degree_prime_parameter}
\leanok
Let $g,\tau\in J_p$, let $T\in k[[X]]$ have order $p$, and
let $r,d$ be natural numbers. Suppose
\[
 D(\tau)=r,\qquad \tau(T)=T,\qquad d<r,\qquad
 v(g(T)-T)=p(d+1).
\]
Then $D(g)=d$. No hypothesis that $T$ is not a $p$th power
is needed.
\end{lemma}

\begin{proof}
First $T$ cannot be a $p$th power. Otherwise its canonical
Frobenius remainder vanishes, so $T=A^p$ with $v(A)=1$.
Since $\tau$ fixes $T$, additivity of Frobenius gives
$(\tau(A)-A)^p=0$, hence $\tau(A)-A=0$ because the
power-series ring is a domain. But $D(\tau)=r<\infty$
means $v(\tau(X)-X)=r+1$. The leading-displacement lemma,
applied to $A$ of order one, gives
$v(\tau(A)-A)=r+1<\infty$, a contradiction.
The decomposition lemma therefore supplies the first nonzero
coefficient $a_u$ whose exponent is not divisible by $p$.
Applying Lemma~\ref{lem:camina-four-depth-equation} to $\tau$
gives $u=r(p-1)+p$.
The finite order of $g(T)-T$ implies $g\ne1$, so its depth
is a finite natural number $n$ and $v(g(X)-X)=n+1$.
Lemma~\ref{lem:camina-five-depth-preservation} now gives $n=d$.

This argument replaces the separability argument for the existence
of $u$ on Camina's p.~106 by the given nontrivial fixing automorphism.
\end{proof}

\begin{lemma}[The lower parameter in uniformizer coordinates]
\label{lem:artin-schreier-base-parameter}
\uses{lem:artin-schreier-weighted-valuation,lem:artin-schreier-laurent-parameter}
\source{camina1997:page-0006}
\lean{Camina1997.ArtinSchreier.baseParameter, Camina1997.ArtinSchreier.powerSeriesEval_baseParameter, Camina1997.ArtinSchreier.order_baseParameter, Camina1997.ArtinSchreier.order_eq_of_weightedOrder_eval}
\leanok
Let $k=\mathbf F_p$, let $K=k((t))$, and let $\gamma\in K$
have order $-m$ with $m>0$ and $p\nmid m$.
Put $L=K[Y]/(Y^p-Y-\gamma)$, write $j:K\hookrightarrow L$
for the coefficient inclusion, and let $w$ be the weighted valuation.
For a uniformizer $\pi\in L$, let
$E_\pi:k[[X]]\simeq\mathcal O_L$ be evaluation at $\pi$.
There is a unique power series $T_\pi$ such that
\[
 E_\pi(T_\pi)=j(t),\qquad v(T_\pi)=p.
\]
Moreover if $w(E_\pi(f))=n$ is a finite natural number, then
$v(f)=n$.
\end{lemma}

\begin{proof}
The weighted valuation satisfies $w(j(t))=pv_t(t)=p\ge0$,
so $j(t)$ belongs to $\mathcal O_L$. Evaluation is a bijection
onto that ring, and its inverse defines $T_\pi$ uniquely.
Evaluation preserves order, so
$v(T_\pi)=w(E_\pi(T_\pi))=w(j(t))=p$.
The same equality $v(f)=w(E_\pi(f))$ proves the last assertion.
If its right-hand side is finite, $E_\pi(f)$ and hence $f$ are
nonzero, so this equality has the ordinary finite-value meaning.
\end{proof}

\begin{lemma}[Actions in uniformizer coordinates]
\label{lem:artin-schreier-uniformizer-action}
\uses{lem:artin-schreier-laurent-parameter,lem:laurent-action-nottingham-restriction,lem:pgroup-power-series-normalization,def:nottingham-depth}
\source{camina1997:page-0005,camina1997:page-0008}
\lean{Camina1997.ArtinSchreier.parameterAction, Camina1997.ArtinSchreier.orderTop_parameterAction, Camina1997.ArtinSchreier.parameterAction_injective_iff, Camina1997.ArtinSchreier.powerSeriesAction, Camina1997.ArtinSchreier.powerSeriesEval_powerSeriesAction, Camina1997.ArtinSchreier.powerSeriesAction_injective_iff, Camina1997.ArtinSchreier.order_powerSeriesAction_sub, Camina1997.ArtinSchreier.nottinghamAction, Camina1997.ArtinSchreier.powerSeriesEval_nottinghamAction, Camina1997.ArtinSchreier.nottinghamAction_injective_iff, Camina1997.ArtinSchreier.depth_nottinghamAction, Camina1997.ArtinSchreier.isClosedEmbedding_nottinghamAction}
\leanok
For $L,w,\pi,E_\pi$ as in the preceding lemma, suppose a group $G$
acts on $L$ by ring automorphisms $\rho(g)$ preserving $w$.
There is an action $\widehat\rho$ on $k[[X]]$ by $k$-algebra
automorphisms such that
\[
 E_\pi(\widehat\rho(g)(f))=\rho(g)(E_\pi(f))
 \qquad(g\in G,\ f\in k[[X]]).
\]
The action $\widehat\rho$ is faithful exactly when $\rho$ is faithful.
If $G$ is a $p$-group, it gives an action $\iota:G\to J_p$ with
the same evaluation identity and faithfulness property. For every
natural number $n$,
\[
 w(\rho(g)(\pi)-\pi)=n+1\quad\Longrightarrow\quad D(\iota(g))=n.
\]
In this $p$-group case, if $G$ is finite and discrete and $\rho$
is faithful, then $\iota$ is a closed topological embedding.
\end{lemma}

\begin{proof}
Let $U_\pi:k((X))\simeq L$ be Laurent evaluation at $\pi$.
Conjugating the field action gives
$\sigma(g)=U_\pi^{-1}\rho(g)U_\pi$ on $k((X))$.
Every ring automorphism fixes $k=\mathbf F_p$, so these are
$k$-algebra automorphisms. The valuation identity for $U_\pi$
and the hypothesis on $\rho$ show that each $\sigma(g)$ preserves
order. Restricting to the power-series ring gives $\widehat\rho$.
Conjugation is a group isomorphism on automorphism groups, and
restriction preserves and reflects faithfulness by
Lemma~\ref{lem:laurent-action-nottingham-restriction}.
This proves the assertion about faithfulness. The definition of
conjugation and the equality of Laurent and power-series evaluation
on $k[[X]]$ give the displayed evaluation identity.

If $G$ is a $p$-group, the power-series action is normalized,
so it factors through $J_p$ without changing any automorphisms
or faithfulness. Since $E_\pi(X)=\pi$, the evaluation identity
and additivity give
\[
 E_\pi(\iota(g)(X)-X)=\rho(g)(\pi)-\pi.
\]
Preservation of order therefore identifies the finite displacement
order on the two sides. Subtracting one gives the depth assertion.
Finally the faithful finite discrete action gives a closed
embedding by Lemma~\ref{lem:pgroup-power-series-normalization}.
\end{proof}

\begin{lemma}[Depth preservation for a compatible Artin--Schreier action]
\label{lem:artin-schreier-depth-of-restriction}
\uses{lem:artin-schreier-base-parameter,lem:artin-schreier-uniformizer-action,lem:artin-schreier-cyclic-extension,lem:artin-schreier-translation-depth,lem:nottingham-depth-fixed-parameter,lem:nottingham-laurent-extension}
\source{camina1997:page-0009,camina1997:page-0010}
\lean{Camina1997.Nottingham.orderTop_toLaurent_sub_parameter, Camina1997.ArtinSchreier.translationPowerSeries_baseParameter, Camina1997.ArtinSchreier.depth_nottinghamAction_of_restriction}
\leanok
Let $k=\mathbf F_p$, let $K=k((t))$, and let $\gamma\in K$
have order $-m$, where $m>0$ and $p\nmid m$.
Put $L=K[y]$ with $y^p-y=\gamma$, and equip it with the weighted
valuation $w$ and the coefficient inclusion $j:K\hookrightarrow L$.
For $c\in k$, let $\tau_c$ fix $K$ and send $y$ to $y+c$.
Choose a uniformizer $\pi$ satisfying
\[
 w(\tau_c(\pi)-\pi)=m+1\qquad(c\in k,\ c\ne0).
\]
Let a $p$-group $G$ act on $L$ by automorphisms $\rho(g)$
preserving $w$. Suppose $q:G\to J_p(t)$ is a homomorphism
such that
\[
 \rho(g)(j(a))=j(\widetilde{q(g)}(a))
 \qquad(g\in G,\ a\in K),
\]
where the tilde denotes Laurent extension.
Let $\iota:G\to J_p(X)$ be the action in the coordinate $X\mapsto\pi$.
For every $g\in G$ and natural number $d$ with $D_t(q(g))=d<m$,
\[
 D_X(\iota(g))=d.
\]
\end{lemma}

\begin{proof}
Let $T_\pi\in k[[X]]$ be the lower parameter, characterized by
$E_\pi(T_\pi)=j(t)$. It has order $p$.
The translation $\tau_1$ has order $p$ and preserves $w$;
its action in the coordinate $X\mapsto\pi$ is therefore a
Nottingham element $\tau$. The assumed displacement of $\pi$
gives $D_X(\tau)=m$. The translation fixes $j(t)$, so the
evaluation identity and injectivity of $E_\pi$ show that
$\tau(T_\pi)=T_\pi$.

Fix $g$ and $d$ as in the statement. The evaluation identity
for the lifted action and the restriction hypothesis give
\[
 E_\pi(\iota(g)(T_\pi)-T_\pi)
   =j(\widetilde{q(g)}(t)-t).
\]
The definition of finite depth gives
$v_t(\widetilde{q(g)}(t)-t)=d+1$. Since $w\circ j=pv_t$,
the evaluated displacement has weight $p(d+1)$.
Evaluation preserves order, hence
\[
 v_X(\iota(g)(T_\pi)-T_\pi)=p(d+1).
\]
Apply Lemma~\ref{lem:nottingham-depth-fixed-parameter} to
$\iota(g)$, $\tau$, and $T_\pi$, using $d<m$. It gives the
required depth equality.
\end{proof}

\begin{theorem}[One central lift preserves all nontrivial quotient depths]
\label{thm:witt-simultaneous-depth-preserving-central-lift}
\uses{thm:witt-reduced-central-lift,lem:artin-schreier-semilinear-valuation,lem:artin-schreier-translation-depth,lem:artin-schreier-uniformizer-action,lem:artin-schreier-laurent-parameter,lem:artin-schreier-depth-of-restriction,lem:nottingham-depth-filtration,lem:nottingham-laurent-isometry}
\source{camina1997:page-0009,camina1997:page-0010}
\lean{Camina1997.Witt.exists_depth_preserving_centralLift}
\leanok
Put $k=\mathbf F_p$ and $K=k((t))$, with its order valuation $v$.
Let $F$ be a field with an isomorphism $e_F:F\simeq K$, and
let $\alpha:F\hookrightarrow K$ make $K/F$ finite Galois.
Write $M=\operatorname{Gal}_{\alpha}(K/F)$.
Let $H$ be a finite $p$-group and let
\[
 1\longrightarrow(k,+)\xrightarrow{\eta}H
   \xrightarrow{q}M\longrightarrow1
\]
be a central extension. Suppose a homomorphism
$\psi:H\to J_p(t)$ satisfies
\[
 \widetilde{\psi(h)}(a)=q(h)(a)
 \qquad(h\in H,\ a\in K).
\]
There is a field embedding $\beta:F\hookrightarrow K$ making
$K/F$ finite Galois, an isomorphism
$\rho:H\simeq\operatorname{Gal}_{\beta}(K/F)$, and an injective
homomorphism $\varphi:H\to J_p$ such that
\[
 v(\rho(h)(a))=v(a),\qquad
 \widetilde{\varphi(h)}(a)=\rho(h)(a)
 \qquad(h\in H,\ a\in K),
\]
and, simultaneously for every $h\in H$,
\[
 \psi(h)\ne1\quad\Longrightarrow\quad
 D(\varphi(h))=D(\psi(h)).
\]
The embeddings $\alpha$ and $\beta$ into the Laurent field may differ;
the same base field $F$ and its specified isomorphism $e_F$ are retained.
\end{theorem}

\begin{proof}
Every nonidentity element of Nottingham has a finite natural depth.
Since $H$ is finite, choose a natural number $B$ bounding the depths
of all $\psi(h)\ne1$; if there are none, take $B=0$.
The reduced central-lift theorem, applied with this single bound,
gives a parameter $r\in K$ of order $-m$, with $m>B$ and
$p\nmid m$, a finite Galois field
\[
 A=K[y],\qquad y^p-y=r,
\]
and an isomorphism $\sigma:H\simeq\operatorname{Gal}(A/F)$
whose restriction to $K$ is $q$. Let $j:K\hookrightarrow A$
be the coefficient inclusion and let $w$ be the weighted valuation.

The automorphism $q(h)$ is the Laurent extension of $\psi(h)$,
so it preserves the order on $K$. The semilinear valuation lemma
therefore gives $w(\sigma(h)(z))=w(z)$ for all $h\in H$ and
$z\in A$. Choose a uniformizer $\pi$ for which every nonzero
translation $y\mapsto y+c$ displaces $\pi$ by weight $m+1$,
as supplied by the translation-depth lemma.
The action in uniformizer coordinates gives a homomorphism
$\varphi:H\to J_p$. It is injective because the full Galois
action $\sigma$ is faithful and the coordinate action preserves
faithfulness.

If $\psi(h)\ne1$, write $D(\psi(h))=d$.
The choice of $B$ gives $d\le B<m$. The restriction formula for
$\sigma$ and the hypothesis on $\psi$ identify its restriction
with $\widetilde{\psi(h)}$. Applying
Lemma~\ref{lem:artin-schreier-depth-of-restriction} gives
$D(\varphi(h))=d$. Both the extension and the uniformizer were
chosen before $h$, so this proves the assertion for all such
$h$ simultaneously.

Finally let $U_\pi:K\simeq A$ be Laurent evaluation at $\pi$.
Define
\[
 \beta=U_\pi^{-1}\circ j\circ\alpha,
 \qquad
 \rho(h)=U_\pi^{-1}\sigma(h)U_\pi.
\]
Then $U_\pi$ identifies the extension through $\beta$ with
$A/F$. It follows that this is finite Galois and that $\rho$
is an isomorphism with its full Galois group. Since $U_\pi$
identifies Laurent order with $w$, the action $\rho$ preserves
order. By the definition of the uniformizer action,
$\widetilde{\varphi(h)}=\rho(h)$ on all of $K$.
Thus the resulting Nottingham embedding and the full Laurent
Galois realization agree, as required.

Camina's Theorem~4 is stated for one nontrivial restricted element.
Taking the maximum of the finitely many lower depths in its
parameter-choice argument gives the simultaneous conclusion above.
The reduced-pole adjustment also allows a central kernel of order
$p$ without a Frattini condition.
\end{proof}

\begin{lemma}[A central prime-order subgroup inside a normal kernel]
\label{lem:finite-p-group-central-prime-step-in-kernel}
\uses{lem:finite-p-group-central-prime-step}
\source{camina1997:page-0009,camina1997:page-0010}
\lean{IsPGroup.exists_central_subgroup_card_prime_le_normal, IsPGroup.exists_central_prime_step_le_normal}
\leanok
Let $H$ be a finite $p$-group and let $K\ne1$ be a normal subgroup.
There is a normal subgroup $N$ of $H$ such that
\[
 N\le K\cap Z(H),\qquad |N|=p.
\]
Moreover, $H/N$ is a $p$-group and $|H/N|<|H|$.
\end{lemma}

\begin{proof}
The group $H$ is nilpotent. Indeed, start its upper central series
with $Z_0=1$ and define $Z_{i+1}/Z_i=Z(H/Z_i)$.
If $Z_i\ne H$, Lemma~\ref{lem:finite-p-group-central-prime-step}
applied to the nontrivial $p$-group $H/Z_i$ shows that its center
is nontrivial. Thus $Z_i<Z_{i+1}$. Finiteness implies that
$Z_c=H$ for some $c$.

Choose the least $i\ge1$ for which $K\cap Z_i\ne1$ and an element
$1\ne x\in K\cap Z_i$. For every $h\in H$, the commutator
$hxh^{-1}x^{-1}$ lies in $K$ by normality and in $Z_{i-1}$ by
the definition of the upper central series. Minimality of $i$
makes this intersection trivial. Consequently $x\in Z(H)$,
so $K\cap Z(H)\ne1$. This intersection is a nontrivial finite
$p$-group; the prime-power subgroup theorem supplies a subgroup
$N$ of order $p$ in it. Since $N$ is central, it is normal in $H$.
Finally, $|H/N|=|H|/p$ is a power of $p$ strictly smaller than
$|H|$, as $p>1$.

This is the group-theoretic refinement needed to iterate the
central lifting step of \cite{Camina1997}, pp.~109--110,
inside the kernel of a prescribed finite quotient.
\end{proof}

\begin{theorem}[Depth-preserving realization over a finite quotient]
\label{thm:witt-depth-preserving-finite-quotient-lift}
\uses{lem:finite-p-group-central-prime-step-in-kernel,thm:witt-simultaneous-depth-preserving-central-lift,lem:nottingham-laurent-isometry}
\source{camina1997:page-0009,camina1997:page-0010,camina1997:page-0011}
\lean{Camina1997.Witt.exists_depth_preserving_lift}
\leanok
Put $k=\mathbf F_p$ and $K=k((t))$, with order valuation $v$.
Let $F$ be a field with a specified isomorphism $e_F:F\simeq K$,
and let $\alpha:F\hookrightarrow K$ make $K/F$ finite Galois.
Let $G$ be finite, and suppose that an isomorphism
$\rho_G:G\simeq\operatorname{Gal}_{\alpha}(K/F)$ and a homomorphism
$\psi_G:G\to J_p$ satisfy
\[
 \widetilde{\psi_G(g)}(a)=\rho_G(g)(a)
 \qquad(g\in G,\ a\in K).
\]
For any finite $p$-group $H$ and surjective homomorphism $q:H\to G$,
there are a field embedding $\beta:F\hookrightarrow K$ making
$K/F$ finite Galois, an isomorphism
$\rho:H\simeq\operatorname{Gal}_{\beta}(K/F)$, and an injective
homomorphism $\varphi:H\to J_p$ such that
\[
 v(\rho(h)(a))=v(a),\qquad
 \widetilde{\varphi(h)}(a)=\rho(h)(a)
 \qquad(h\in H,\ a\in K),
\]
and, simultaneously for all $h\in H$,
\[
 \psi_G(q(h))\ne1\quad\Longrightarrow\quad
 D(\varphi(h))=D(\psi_G(q(h))).
\]
The base field $F$ and its specified isomorphism $e_F$ are retained,
while its embedding into $K$ may change.
\end{theorem}

\begin{proof}
First, $\psi_G$ is injective: equality of two of its values gives
equality of their Laurent extensions, hence of their images under
$\rho_G$, and then equality in $G$.
We use strong induction on $|H|$, keeping the initial realization
of $G$ fixed. If $q$ is injective, it is an isomorphism.
Take $\beta=\alpha$, $\rho=\rho_G\circ q$, and
$\varphi=\psi_G\circ q$. These maps have the asserted
faithfulness and agreement. Each $\rho(h)$ preserves order because
it is the Laurent extension of a Nottingham element; the depth
equalities are identities.

Suppose that $q$ is not injective. Its kernel is a nontrivial
normal subgroup. Choose $N\le\ker q\cap Z(H)$ of order $p$
by Lemma~\ref{lem:finite-p-group-central-prime-step-in-kernel}.
Let $r:H\to Q=H/N$ be the quotient map. Since $N\le\ker q$,
there is a surjective homomorphism $\bar q:Q\to G$ satisfying
$q=\bar q\circ r$. The group $Q$ is a $p$-group with $|Q|<|H|$.
The induction hypothesis supplies an embedding
$\alpha_Q:F\hookrightarrow K$, its full finite Galois
realization $\rho_Q:Q\simeq\operatorname{Gal}_{\alpha_Q}(K/F)$,
and an injective homomorphism $\varphi_Q:Q\to J_p$ agreeing
with that action on $K$. They satisfy
\[
 \psi_G(\bar q(z))\ne1\quad\Longrightarrow\quad
 D(\varphi_Q(z))=D(\psi_G(\bar q(z)))
 \qquad(z\in Q).
\]

The surjection $s=\rho_Q\circ r$ has kernel $N$.
A group of order $p$ is cyclic, so identify $N$ with $(k,+)$.
This gives a central extension of
$\operatorname{Gal}_{\alpha_Q}(K/F)$ by $(k,+)$.
Set $\psi=\varphi_Q\circ r$. Its Laurent action is $s$,
by the agreement assertion for $\varphi_Q$.
Apply Theorem~\ref{thm:witt-simultaneous-depth-preserving-central-lift}
to this central extension, using the same $F$ and $e_F$.
It produces $\beta$, $\rho$, and $\varphi$ with the full
Galois, valuation, faithfulness, and agreement assertions of
the conclusion, and with $D(\varphi(h))=D(\psi(h))$
whenever $\psi(h)\ne1$.

If $\psi_G(q(h))\ne1$, then $r(h)\ne1$, since
$q=\bar q\circ r$. Injectivity of $\varphi_Q$ therefore
gives $\psi(h)\ne1$. The central-step equality and the induction
hypothesis now give
\[
 D(\varphi(h))=D(\varphi_Q(r(h)))
             =D(\psi_G(\bar q(r(h))))
             =D(\psi_G(q(h))).
\]
Both hypotheses apply to the same constructed action, so these
equalities hold for every indicated $h$ simultaneously.

This extends the central step in Camina's Theorem~4 to an arbitrary
finite $p$-group quotient by induction inside its kernel. It supplies
the depth assertion needed between successive finite groups in
the tower of Theorem~5. The assertion compares depths in Laurent
coordinates and does not assert equality of coefficients between
successive realizations.
\end{proof}
