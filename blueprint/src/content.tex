\section{Topological groups and closed embeddings}

Fix a prime number $p$. All homomorphisms described as continuous are
continuous for the stated group topologies. The definitions in this section
make explicit the topological conventions used in
\cite{Camina1997}.

\begin{definition}[Pro-$p$ group]
\label{def:pro-p-group}
\lean{Camina1997.IsProPGroup}
\leanok
A pro-$p$ group is a compact Hausdorff topological group $P$ with a
neighbourhood basis at the identity consisting of open normal subgroups
$N$ for which $[P:N]$ is a finite power of $p$.
\end{definition}

\begin{definition}[Countable base]
\label{def:countably-based}
\notready
A topological group is countably based if its identity has a countable
neighbourhood basis. Thus there are open identity neighbourhoods
$U_1,U_2,\ldots$ such that every identity neighbourhood contains some
$U_n$.
\end{definition}

\begin{definition}[Finite generation]
\label{def:topologically-finitely-generated}
\notready
A topological group $P$ is topologically finitely generated when some
finite subset $S\subseteq P$ generates a dense subgroup of $P$.
\end{definition}

\begin{definition}[Closed group embedding]
\label{def:closed-group-embedding}
\notready
A closed group embedding $f:P\longrightarrow G$ is a group homomorphism
which is a homeomorphism from $P$ onto its image with the subspace
topology, and whose image is closed in $G$.
\end{definition}

\begin{lemma}[Compact source and Hausdorff target]
\label{lem:compact-injection-closed-embedding}
\uses{def:closed-group-embedding}
\source{camina1997:page-0012}
\notready
Let $P$ be a compact topological group and $G$ a Hausdorff topological
group. Every continuous injective homomorphism
$f:P\longrightarrow G$ is a closed group embedding.
\end{lemma}

\begin{proof}
First, a compact subset $K$ of a Hausdorff space is closed. Indeed, if
$y\notin K$, separate $y$ and each $x\in K$ by disjoint open
neighbourhoods $V_x$ and $W_x$, with $x\in V_x$ and $y\in W_x$.
Finitely many $V_{x_1},\ldots,V_{x_r}$ cover $K$. Their corresponding
intersection $W_{x_1}\cap\cdots\cap W_{x_r}$ is an open neighbourhood
of $y$ disjoint from $K$. The empty compact set is closed as well.

For every closed subset $C$ of $P$, compactness of $P$ implies compactness
of $C$: add $P\setminus C$ to any open cover of $C$, then restrict a
finite subcover of $P$. Continuity of $f$ implies that $f(C)$ is compact,
because the inverse images of an open cover of $f(C)$ cover $C$.
The preceding paragraph therefore shows that $f(C)$ is closed in $G$,
and hence in $f(P)$. The map $P\longrightarrow f(P)$ is a continuous
bijection by injectivity of $f$; the closed-set assertion just proved
shows that its inverse is continuous. Taking $C=P$ also proves that
$f(P)$ is closed in $G$. Finally, the image of a homomorphism is a
subgroup, so all the requirements of a closed group embedding hold.
\end{proof}

This lemma states the compactness argument at the end of Camina's proof
of Theorem~5 in a form independent of the Nottingham group; the proof
above supplies the elementary topological details.

\section{The Nottingham group and the universality target}

Write $k=\mathbf F_p$, $R=k[[t]]$, and $\mathfrak m=tR$. More generally,
the algebraic assertions below hold for any field $k$. Multiplication of
automorphisms means $(\sigma\tau)(f)=\sigma(\tau(f))$.
Write $A=\operatorname{Aut}_{k\text{-alg}}(R)$.
This is the ring-automorphism description in the introduction of
\cite{Camina1997}; the Laurent-series description on the next page is
related to it by the extension-and-restriction statement below.

\begin{lemma}[Constant coefficients and continuity]
\label{lem:power-series-aut-continuity}
\source{camina1997:page-0001,camina1997:page-0002}
\lean{PowerSeries.constantCoeff_algEquiv_X, PowerSeries.constantCoeff_algEquiv, PowerSeries.continuous_algEquiv}
\leanok
Every $k$-algebra automorphism $\sigma$ of $R$ satisfies
$[t^0]\sigma(t)=0$ and $[t^0]\sigma(f)=[t^0]f$ for every $f\in R$.
It and its inverse are continuous for the $\mathfrak m$-adic topology.
\end{lemma}

\begin{proof}
A series $f=\sum a_it^i$ is a unit exactly when $a_0\ne0$.
Necessity follows by taking constant coefficients of an inverse equation.
For sufficiency define $b_0=a_0^{-1}$ and, for $n>0$, define
$b_n=-a_0^{-1}\sum_{i=1}^n a_i b_{n-i}$. The product of $f$ with
$\sum b_nt^n$ has constant coefficient $1$ and all other coefficients
zero, and commutativity gives a two-sided inverse.
Thus $\mathfrak m$ is precisely the set of nonunits. Automorphisms and
their inverses preserve units, so they preserve $\mathfrak m$; in
particular $[t^0]\sigma(t)=0$.
Write $f=tg+a_0$, where $g=\sum_{i\ge0}a_{i+1}t^i$.
Applying $\sigma$ and taking constant coefficients gives
$[t^0]\sigma(f)=[t^0]\sigma(t)[t^0]\sigma(g)+a_0=a_0$.

The preservation of $\mathfrak m$ implies preservation of every
$\mathfrak m^n$, since an automorphism preserves sums and products.
A basic neighbourhood of $\sigma(f)$ is
$\sigma(f)+\mathfrak m^n$, and its inverse image is
$f+\mathfrak m^n$. Hence $\sigma$ is continuous. Applying the same
argument to $\sigma^{-1}$ proves continuity of the inverse.
\end{proof}

\begin{lemma}[Linear coefficient of an automorphism]
\label{lem:power-series-aut-linear-coeff}
\uses{lem:power-series-aut-continuity}
\source{camina1997:page-0002}
\lean{PowerSeries.coeff_one_algEquiv}
\leanok
For every $k$-algebra automorphism $\sigma$ of $R$ and every $f\in R$,
\[
 [t]\sigma(f)=[t]\sigma(t)\,[t]f.
\]
\end{lemma}

\begin{proof}
Write $f=tg+a_0$ as above, so $[t^0]g=[t]f$.
The linear coefficient of a product $uv$ is
$[t]u[t^0]v+[t^0]u[t]v$. Since $\sigma$ fixes constants,
$\sigma(f)=\sigma(t)\sigma(g)+a_0$. The preceding lemma gives
$[t^0]\sigma(t)=0$ and $[t^0]\sigma(g)=[t^0]g$.
Substitution in the product formula proves the assertion.
\end{proof}

\begin{definition}[Normalization]
\label{def:normalized-aut}
\source{camina1997:page-0001,camina1997:page-0002}
\lean{PowerSeries.IsNormalized}
\leanok
A $k$-algebra automorphism $\sigma$ of $R$ is normalized if
$[t]\sigma(t)=1$.
\end{definition}

\begin{lemma}[Equivalent normalization conditions]
\label{lem:normalization-iff}
\uses{def:normalized-aut,lem:power-series-aut-linear-coeff}
\source{camina1997:page-0001,camina1997:page-0002}
\lean{PowerSeries.isNormalized_iff_X_sq_dvd, PowerSeries.isNormalized_iff_forall_X_sq_dvd, PowerSeries.isNormalized_iff_forall_X_dvd}
\leanok
For a $k$-algebra automorphism $\sigma$ of $R$, the following conditions
are equivalent:
\begin{enumerate}
\item $\sigma$ is normalized;
\item $\sigma(t)-t\in\mathfrak m^2$;
\item $\sigma(f)-f\in\mathfrak m^2$ for every $f\in R$;
\item $\sigma(f)-f\in\mathfrak m^2$ for every $f\in\mathfrak m$.
\end{enumerate}
In particular normalization means that the induced action on
$\mathfrak m/\mathfrak m^2$ is the identity.
\end{lemma}

\begin{proof}
A series is in $t^2R$ exactly when its constant and linear
coefficients vanish: dividing the tail by $t^2$ proves sufficiency,
and the product formula proves necessity.
The constant coefficient of $\sigma(t)-t$ is zero, and its linear
coefficient is $[t]\sigma(t)-1$. This proves (1)$\Leftrightarrow$(2).
For arbitrary $f$, constant coefficients of $\sigma(f)-f$ vanish.
Under (1), the linear-coefficient formula also gives
$[t](\sigma(f)-f)=(1-1)[t]f=0$, proving (3).
Condition (3) implies (4) by restriction, and (4) implies (2) by
setting $f=t$. Preservation of $\mathfrak m$ and $\mathfrak m^2$
makes the quotient action well-defined, and (4) says exactly that
this action is the identity.
\end{proof}

\begin{lemma}[Closure of normalization]
\label{lem:normalized-aut-closure}
\uses{def:normalized-aut,lem:power-series-aut-linear-coeff}
\source{camina1997:page-0002}
\lean{PowerSeries.isNormalized_one, PowerSeries.IsNormalized.mul, PowerSeries.IsNormalized.inv}
\leanok
The identity is normalized. The composite of two normalized
$k$-algebra automorphisms is normalized, and the inverse of a normalized
$k$-algebra automorphism is normalized.
\end{lemma}

\begin{proof}
The identity sends $t$ to $t$, whose linear coefficient is $1$.
The linear-coefficient formula gives
$[t](\sigma\tau)(t)=[t]\sigma(t)[t]\tau(t)=1$ for normalized
$\sigma,\tau$. Finally apply that formula to $\sigma$ and
$f=\sigma^{-1}(t)$:
$1=[t]t=[t]\sigma(t)[t]\sigma^{-1}(t)$.
If $\sigma$ is normalized this equality reads
$1=[t]\sigma^{-1}(t)$, proving closure under inverse.
\end{proof}

\begin{definition}[The Nottingham group]
\label{def:nottingham-group}
\uses{lem:normalized-aut-closure,lem:normalization-iff,lem:power-series-aut-continuity}
\source{camina1997:page-0001,camina1997:page-0002}
\lean{PowerSeries.normalizedAutSubgroup, Camina1997.Nottingham}
\leanok
For a prime $p$, the Nottingham group $J_p$ is the subgroup of
$\operatorname{Aut}_{\mathbf F_p\text{-alg}}(\mathbf F_p[[t]])$
consisting of normalized automorphisms, with multiplication given by
composition. Equivalently, it is precisely the group of continuous
$\mathbf F_p$-algebra automorphisms with continuous inverses satisfying
$\sigma(t)-t\in t^2\mathbf F_p[[t]]$.
\end{definition}

\begin{lemma}[Extension of automorphisms to Laurent series]
\label{lem:nottingham-laurent-extension}
\uses{def:nottingham-group,lem:nottingham-congruence-filtration}
\source{camina1997:page-0001,camina1997:page-0002}
\lean{Camina1997.Nottingham.toLaurent, Camina1997.Nottingham.toLaurent_algebraMap, Camina1997.Nottingham.toLaurent_injective, Camina1997.Nottingham.toLaurent_unique, Camina1997.Nottingham.toLaurent_congruence_iff}
\leanok
Every element $\sigma\in J_p$ extends uniquely to a
$\mathbf F_p$-algebra automorphism of $K=\mathbf F_p((t))$.
Extension is an injective group homomorphism and, for every $n\ge0$,
\[
 \sigma(t)-t\in t^{n+1}\mathbf F_p[[t]]\quad\Longleftrightarrow\quad
 \widetilde\sigma(t)-t\in t^{n+1}\mathbf F_p[[t]],
\]
where the right side uses the standard inclusion of power series
in Laurent series.
\end{lemma}

\begin{proof}
Every Laurent series is a fraction of power series: multiplication
by a sufficiently large power of $t$ removes its negative exponents.
The map $a/b\mapsto\sigma(a)/\sigma(b)$ is well-defined, since equality
of two fractions is equality of their cross-products and $\sigma$
preserves multiplication and nonzero denominators. The fraction
formulas for sum and product prove that it is an algebra homomorphism.
Extension of $\sigma^{-1}$ is a two-sided inverse. Any extension must
have this fraction formula, proving uniqueness. Applying the same
formula to a composite proves multiplicativity of extension.
An equality of extensions restricts to an equality on power series,
proving injectivity. Finally the inclusion of power series is injective
and extension agrees with $\sigma$ on $t$, so divisibility by $t^{n+1}$
in the power-series ring is precisely the displayed condition.
\end{proof}

\begin{lemma}[The Laurent extension preserves the valuation]
\label{lem:nottingham-laurent-isometry}
\uses{lem:nottingham-laurent-extension,lem:aut-congruence-parameter}
\source{camina1997:page-0001,camina1997:page-0002}
\notready
Let $v$ be the multiplicative $t$-adic valuation on
$K=\mathbf F_p((t))$, with $v(t)<1$ and $v(0)=0$.
For every $\sigma\in J_p$ and $x\in K$,
$v(\widetilde\sigma(x))=v(x)$.
Consequently $\widetilde\sigma$ and its inverse are uniformly
continuous for the valuation uniformity, and form a homeomorphism of $K$.
\end{lemma}

\begin{proof}
For $f\in R=\mathbf F_p[[t]]$ and every integer $n\ge0$,
$t^n\mid f$ if and only if $t^n\mid\sigma(f)$: forward divisibility
follows because $\sigma(t)$ has order one, and the reverse implication
follows by applying $\sigma^{-1}$.
For nonzero $f$ its order is the largest $n$ with $t^n\mid f$,
so $\sigma(f)$ has the same order and the same valuation. Zero is
preserved as well. Write any $x\in K$ as $a/b$ with $a,b\in R$
and $b\ne0$. The extension formula and multiplicativity of the valuation
give
\[
 v(\widetilde\sigma(x))
 =v(\sigma(a))/v(\sigma(b))=v(a)/v(b)=v(x).
\]
Because the extension preserves subtraction, it also preserves
$v(x-y)$. A basis of entourages for the valuation uniformity consists
of the sets of pairs with $v(x-y)<\gamma$, for nonzero valuation
radii $\gamma$. The inverse image of each such entourage is itself,
which proves uniform continuity. The inverse is the extension of
$\sigma^{-1}$ and satisfies the same argument. The two inverse
continuous maps give the stated homeomorphism.
\end{proof}

\begin{lemma}[Extension and restriction to Laurent series]
\label{lem:nottingham-laurent-equivalence}
\uses{lem:nottingham-laurent-isometry}
\source{camina1997:page-0001,camina1997:page-0002}
\notready
Put $K=\mathbf F_p((t))$, with its valuation topology, and identify
$R=\mathbf F_p[[t]]$ with its valuation ring. Extension to fractions
is a group isomorphism from $J_p$ to the group of continuous
$\mathbf F_p$-algebra automorphisms $\gamma$ of $K$ with
$\gamma(t)-t\in t^2R$. Its inverse is restriction to $R$.
For every $n\ge1$ this isomorphism preserves the condition
$\sigma(t)-t\in t^{n+1}R$.
\end{lemma}

\begin{proof}
Each nonzero series has the form $t^r u$ with $r\ge0$ and $u$ a
unit, by taking its first nonzero coefficient. Consequently every
Laurent series is a fraction of power series, and an automorphism
$\sigma$ of $R$ has a unique extension
$\widetilde\sigma(a/b)=\sigma(a)/\sigma(b)$ to $K$.
The fraction formula is independent of the representation because
$a/b=c/d$ implies $ad=bc$, and applying $\sigma$ preserves this
equality. The extension of $\sigma^{-1}$ is its inverse.
For normalized $\sigma$, write $\sigma(t)=tv$ with $v\in1+tR$ a
unit. The image of $t^r u$, for $r\in\mathbf Z$ and $u\in R^\times$,
has valuation $r$. Thus the extension preserves valuation and is
continuous, and has the asserted normalization.

Conversely, suppose $\gamma$ is such a continuous automorphism of
$K$. Put $z=\gamma(t)$, which has valuation one by normalization.
For $f=\sum_{i\ge0}a_it^i\in R$, its polynomial partial sums
converge to $f$. Continuity and fixation of the coefficients imply
$\gamma(f)=\sum_{i\ge0}a_i z^i$. This sum belongs to $R$: each term
has nonnegative valuation, and the tails have valuations tending
to infinity. If $a_0\ne0$, the image also has nonzero constant
coefficient $a_0$. Writing a nonzero Laurent series as $t^r u$
therefore shows that $\gamma$ preserves valuation. Surjectivity on
$K$ now gives surjectivity on $R$, since a preimage of an element
of nonnegative valuation also has nonnegative valuation.
Hence restriction gives an automorphism of $R$, which is normalized.
The two constructions are inverse by the fraction formula. The same
formula proves compatibility with composition. Finally the image of
$t$ is unchanged by extension or restriction, so every stated
congruence condition is preserved.
\end{proof}

\begin{lemma}[Congruence determined by the parameter]
\label{lem:aut-congruence-parameter}
\uses{lem:power-series-aut-continuity}
\source{camina1997:page-0002}
\lean{PowerSeries.X_pow_dvd_algEquiv_iff, PowerSeries.X_pow_dvd_algEquiv_sub_iff}
\leanok
Let $\sigma,\tau\in A$ and $r\ge0$. Then
$\sigma(t)-\tau(t)\in t^rR$ if and only if
$\sigma(f)-\tau(f)\in t^rR$ for every $f\in R$.
Every automorphism of $R$ preserves each ideal $t^rR$.
\end{lemma}

\begin{proof}
Preservation follows because $\sigma(t)\in tR$, hence
$\sigma(t^r h)=\sigma(t)^r\sigma(h)\in t^rR$; applying the same
argument to the inverse gives equality of ideals.
For the forward implication use induction on $r$. The case $r=0$
is immediate. Suppose the parameter difference is in $t^{r+1}R$.
Write $f=th+a$ with $a\in k$. Then
\[
\sigma(f)-\tau(f)=(\sigma(t)-\tau(t))\sigma(h)
                   +\tau(t)(\sigma(h)-\tau(h)).
\]
The first term lies in $t^{r+1}R$. The induction hypothesis, applied
to the weaker parameter congruence modulo $t^r$, places the second
factor in the second term in $t^rR$. Since $\tau(t)\in tR$, that
term also lies in $t^{r+1}R$. The reverse implication takes $f=t$.
\end{proof}

\begin{lemma}[Realization of normalized parameters]
\label{lem:nottingham-parameters}
\uses{def:nottingham-group,lem:aut-congruence-parameter}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.coefficientEquiv, Camina1997.Nottingham.coeff_ofCoefficients_X}
\leanok
For every sequence $(a_i)_{i\ge0}$ in $\mathbf F_p$, there is a unique
$\sigma\in J_p$ such that
$\sigma(t)=t+\sum_{i\ge0}a_i t^{i+2}$.
Thus the map $\sigma\mapsto([t^{i+2}]\sigma(t))_{i\ge0}$ is a bijection
from $J_p$ to $\mathbf F_p^{\mathbf N}$.
\end{lemma}

\begin{proof}
Put $P=t+\sum_{j\ge2}b_jt^j$, with $b_j=a_{j-2}$.
Substitution $f\mapsto f(P)$ is well-defined since $P$ has zero constant
coefficient: each coefficient of the substituted series involves only
finitely many terms. The same finiteness proves preservation of addition,
multiplication, constants, and associativity of substitution.
Construct $Q=t+\sum_{n\ge2}c_nt^n$ recursively, choosing
\[
 c_n=-\sum_{j=2}^{n}b_j[t^n]
       \left(t+\sum_{i=2}^{n-1}c_it^i\right)^j.
\]
For $j\ge2$, the coefficient in this expression depends only on the
coefficients of $Q$ of degree less than $n$. Hence the coefficient of
$t^n$ in $P(Q)$ is zero, while the constant and linear coefficients
are $0$ and $1$. This proves $P(Q)=t$.
Apply the same construction to $Q$ to get $S$ with $Q(S)=t$.
Associativity gives $P=P(Q(S))=(P(Q))(S)=S$, so also $Q(P)=t$.
Substitution by $P$ and $Q$ are therefore mutually inverse algebra
homomorphisms, and substitution by $P$ is normalized.
If two automorphisms have the same parameter, the preceding congruence
lemma makes their values on every series congruent modulo every power
of $t$, hence equal coefficient by coefficient. This proves uniqueness
and the asserted bijection.
\end{proof}

\begin{lemma}[The congruence filtration]
\label{lem:nottingham-congruence-filtration}
\uses{def:nottingham-group,lem:aut-congruence-parameter}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.congruenceSubgroup, Camina1997.Nottingham.mem_congruenceSubgroup_iff, Camina1997.Nottingham.congruenceSubgroup_normal, Camina1997.Nottingham.congruenceSubgroup_antitone, Camina1997.Nottingham.congruenceSubgroup_zero, Camina1997.Nottingham.congruenceSubgroup_one, Camina1997.Nottingham.iInf_congruenceSubgroup}
\leanok
For $n\ge0$ the sets
$J_{p,n}=\{\sigma\in J_p:\sigma(t)-t\in t^{n+1}R\}$
are normal subgroups. They decrease with $n$, satisfy
$J_{p,0}=J_{p,1}=J_p$, and have intersection $\{1\}$.
\end{lemma}

\begin{proof}
By the preceding lemma membership means that $\sigma$ fixes every
series modulo $t^{n+1}R$. The identity has this property, and
\[
 (\sigma\tau)(f)-f=(\sigma(\tau(f))-\tau(f))+(\tau(f)-f)
\]
proves closure under products. Setting $h=\sigma^{-1}(f)$ in
$\sigma(h)-h\in t^{n+1}R$ and negating proves closure under inverses.
For $\gamma\in J_p$, the difference
$(\gamma\sigma\gamma^{-1})(f)-f$
is $\gamma(\sigma(h)-h)$, where $h=\gamma^{-1}(f)$, and hence lies
in the same ideal. This proves normality. Inclusion of the powers
of $t$ gives the decreasing property. Constant-coefficient
preservation gives $J_{p,0}=J_p$ and normalization gives
$J_{p,1}=J_p$.
Finally, if $\sigma$ belongs to every subgroup, then for every $f$
and every $i$ the coefficient of degree $i$ in $\sigma(f)-f$
vanishes, by divisibility by $t^{i+1}$. Thus $\sigma(f)=f$ for all
$f$, so $\sigma=1$.
\end{proof}

\begin{lemma}[The congruence topology is Hausdorff]
\label{lem:nottingham-hausdorff-group}
\uses{def:nottingham-congruence,lem:nottingham-congruence-filtration}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.instIsTopologicalGroup, Camina1997.Nottingham.instT2Space, Camina1997.Nottingham.instFirstCountableTopology, Camina1997.Nottingham.isOpen_congruenceSubgroup}
\leanok
The congruence topology makes $J_p$ a Hausdorff topological group
with a countable neighbourhood basis. Every $J_{p,n}$ is open.
\end{lemma}

\begin{proof}
The nested cosets form a basis: if two such cosets contain $\sigma$,
then $\sigma J_{p,k}$, with $k$ the larger of their indices,
lies in their intersection. Normality implies
\[
 (\sigma J_{p,n})(\tau J_{p,n})=\sigma\tau J_{p,n},
 \qquad (\sigma J_{p,n})^{-1}=\sigma^{-1}J_{p,n}.
\]
These identities prove continuity of multiplication and inversion.
At any $\sigma\in J_{p,n}$ the coset $\sigma J_{p,n}$ is precisely
$J_{p,n}$, proving its openness. The displayed basis is countable.

For distinct $\sigma,\tau$, separation by the filtration supplies
$n$ such that $\sigma^{-1}\tau\notin J_{p,n}$.
Their cosets of $J_{p,n}$ are disjoint: membership of a point in both
would give $\sigma^{-1}\tau\in J_{p,n}$ by the subgroup laws.
They are open neighbourhoods, proving the Hausdorff property.
\end{proof}

\begin{lemma}[Order of the congruence quotients]
\label{lem:nottingham-quotient-card}
\uses{lem:nottingham-parameters,lem:nottingham-congruence-filtration}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.card_quotient}
\leanok
For every $n\ge1$ the quotient $J_p/J_{p,n}$ has $p^{n-1}$ elements.
\end{lemma}

\begin{proof}
Two automorphisms $\sigma,\tau$ have the same coset exactly when
$\sigma^{-1}\tau$ fixes $t$ modulo $t^{n+1}$.
Apply $\sigma$, which preserves $t^{n+1}R$, to see that this is
necessary and sufficient for $\tau(t)-\sigma(t)\in t^{n+1}R$.
Both parameters have constant coefficient zero and linear coefficient
one. Thus the condition is equality of their coefficients of degrees
$2,\ldots,n$. The map taking those $n-1$ coefficients consequently
induces an injection from the quotient into $\mathbf F_p^{n-1}$.
It is onto: extend any finite coefficient tuple by zeros and realize
the resulting sequence by the parameter lemma. This is a bijection,
and there are $p$ independent choices for each of the $n-1$ entries.
\end{proof}

\begin{lemma}[Compactness of the Nottingham group]
\label{lem:nottingham-compact}
\uses{lem:nottingham-parameters,lem:nottingham-hausdorff-group}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.coefficientHomeomorph, Camina1997.Nottingham.instCompactSpace}
\leanok
The coefficient bijection $J_p\longrightarrow\mathbf F_p^{\mathbf N}$
is a homeomorphism for the congruence topology and the product of
discrete topologies. In particular, $J_p$ is compact.
\end{lemma}

\begin{proof}
A coset $\sigma J_{p,n}$ consists exactly of the automorphisms whose
parameters have the same coefficients of degrees $2,\ldots,n$ as
$\sigma(t)$: apply $\sigma$ to the defining congruence for
$\sigma^{-1}\tau$. Under the coefficient bijection this is the cylinder
fixing the first $n-1$ coordinates. Such cylinders form a basis of the
product topology, since every finite set of coordinate indices is
contained in an initial segment. Both the bijection and its inverse
therefore take neighbourhood bases to neighbourhood bases, proving the
homeomorphism. Each finite discrete factor is compact, so the product
is compact by the product theorem for compact spaces. A homeomorphism
preserves compactness, proving the final assertion.
\end{proof}

\begin{lemma}[Depth and the congruence filtration]
\label{lem:nottingham-depth-filtration}
\uses{def:nottingham-depth,lem:nottingham-congruence-filtration,lem:aut-congruence-parameter}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.one_le_depth, Camina1997.Nottingham.depth_eq_top_iff, Camina1997.Nottingham.mem_congruenceSubgroup_iff_le_depth, Camina1997.Nottingham.depth_eq_nat_iff, Camina1997.Nottingham.depth_inv, Camina1997.Nottingham.min_depth_le_depth_mul}
\leanok
Every $g\in J_p$ has $D(g)\ge1$, and $D(g)=\infty$ if and only if
$g=1$. Moreover, for $n\ge0$, $g\in J_{p,n}$ if and only if
$n\le D(g)$. Thus $D(g)=n$ if and only if
$g\in J_{p,n}\setminus J_{p,n+1}$.
Depth satisfies $D(g^{-1})=D(g)$ and
$D(gh)\ge\min\{D(g),D(h)\}$.
\end{lemma}

\begin{proof}
Normalization makes the parameter difference divisible by $t^2$.
For a nonzero difference, its order is the least index of a nonzero
coefficient, so it is at least two, and subtracting one gives a
positive integer. The difference is zero exactly when $g(t)=t$,
which determines the identity automorphism by congruence on the parameter.
Divisibility by $t^{n+1}$ is exactly the inequality
$n+1\le\operatorname{ord}_t(g(t)-t)$, hence the stated filtration
criterion and the characterization of equality. Subgroup closure
under inverse gives equal membership of $g$ and $g^{-1}$ in every
$J_{p,n}$, proving equality of their depths. Subgroup closure under
products shows that every integer bounded by both $D(g)$ and $D(h)$
is bounded by $D(gh)$; this is the stated minimum inequality,
including infinite depths.
\end{proof}

\begin{lemma}[The Nottingham group is pro-$p$]
\label{lem:nottingham-pro-p}
\uses{def:pro-p-group,lem:nottingham-hausdorff-group,lem:nottingham-compact,lem:nottingham-quotient-card}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.instIsProPGroup, Camina1997.Nottingham.instFirstCountableTopology, Camina1997.Nottingham.isPGroup_quotient_openNormalSubgroup}
\leanok
The Nottingham group $J_p$ is a countably based pro-$p$ group.
Every open normal subgroup $U$ of $J_p$ has a finite $p$-group quotient.
\end{lemma}

\begin{proof}
The congruence topology is a compact Hausdorff group topology, and
its countable identity basis consists of open normal subgroups
$J_{p,n}$ of index $p^{n-1}$ for $n\ge1$. This is the definition
of a countably based pro-$p$ group. For any open normal subgroup $U$,
choose $n\ge1$ with $J_{p,n}\subseteq U$ using the identity basis.
The natural map $J_p/J_{p,n}\longrightarrow J_p/U$ is onto.
Its source is a finite group of $p$-power order; its target is a
quotient of that group and therefore also a finite $p$-group.
\end{proof}

\begin{lemma}[The congruence inverse limit]
\label{lem:nottingham-inverse-limit}
\uses{lem:nottingham-congruence-filtration,lem:nottingham-compact,lem:nottingham-quotient-card,lem:nottingham-hausdorff-group}
\source{camina1997:page-0002}
\lean{Camina1997.Nottingham.congruenceLimitEquiv, Camina1997.Nottingham.congruenceLimitEquiv_apply, Camina1997.Nottingham.existsUnique_of_compatible_quotients}
\leanok
For $m\ge n$, let
\[\pi_{m,n}:J_p/J_{p,m}\longrightarrow J_p/J_{p,n}\]
be the natural quotient map. The map
\[
 J_p\longrightarrow
 \left\{(x_n)\in\prod_{n\ge0}J_p/J_{p,n}:
       \pi_{m,n}(x_m)=x_n\text{ for all }m\ge n\right\},
 \qquad g\longmapsto(gJ_{p,n})_n,
\]
is an isomorphism of topological groups, where each finite quotient
has the discrete topology and the right side has the subspace
topology of the product.
\end{lemma}

\begin{proof}
The map is a homomorphism because its coordinates are quotient
homomorphisms. It is continuous: the inverse image of a singleton in
each coordinate is an open congruence coset, and product continuity
is tested on coordinates. Its values are compatible by construction.
Its kernel is the intersection of the congruence subgroups, which is
trivial, so it is injective.

For a compatible tuple $(x_n)$, let $C_n$ be its coset in $J_p$.
These sets are nonempty, and compatibility gives $C_m\subseteq C_n$
when $m\ge n$. Each $C_n$ is closed: its complement is the union
of the other open cosets of $J_{p,n}$. Every finite intersection
contains the coset with largest index, hence is nonempty.
Compactness gives a point in the intersection of all $C_n$.
That point maps to $(x_n)$, proving surjectivity. The target is
Hausdorff as a subspace of a product of finite discrete spaces.
A continuous bijection from a compact space to a Hausdorff space is
a homeomorphism: images of closed sets are compact and hence closed.
This proves continuity of the inverse and the assertion.
\end{proof}

\begin{definition}[Closed universality property]
\label{def:closed-universality}
\uses{def:pro-p-group,def:countably-based,def:closed-group-embedding,def:nottingham-group,lem:nottingham-hausdorff-group}

property: for every countably based pro-$p$ group $P$, there exists a
closed group embedding $P\longrightarrow G$.
The universality assertion of Camina's Corollary~2 is
$\mathcal U_p(J_p)$, for each prime $p$.
\end{definition}

This definition specifies the full countably based assertion. It does
not assert that the property holds, and topological finite generation

\section{The conditional reduction}

\begin{theorem}[Reduction through finitely generated pro-$p$ groups]
\label{thm:countable-reduction}
\uses{def:pro-p-group,def:countably-based,def:topologically-finitely-generated,def:closed-universality,lem:compact-injection-closed-embedding}
\source{camina1997:page-0012}
\notready
Let $G$ be a Hausdorff topological group. Suppose that both of the
following hypotheses hold:
\begin{enumerate}
\item Every topologically finitely generated pro-$p$ group $Q$ admits
a continuous injective homomorphism $Q\longrightarrow G$.
\item For every countably based pro-$p$ group $P$, there exist a
topologically finitely generated pro-$p$ group $Q$ and a continuous
injective homomorphism $P\longrightarrow Q$.
\end{enumerate}
Then $\mathcal U_p(G)$ holds.
\end{theorem}

\begin{proof}
Let $P$ be a countably based pro-$p$ group. The second hypothesis
provides a topologically finitely generated pro-$p$ group $Q$ and a
continuous injective homomorphism $i:P\longrightarrow Q$. The first
hypothesis then provides a continuous injective homomorphism
$j:Q\longrightarrow G$. The composite $j\circ i$ is a homomorphism:
for $x,y\in P$,
\[
 j(i(xy))=j(i(x)i(y))=j(i(x))j(i(y)).
\]
It is continuous as a composite of continuous maps, and injective
because equality of its values first implies equality under $i$ by
injectivity of $j$, and then equality in $P$ by injectivity of $i$.
The group $P$ is compact by the definition of a pro-$p$ group, and $G$
is Hausdorff by assumption. The compact-source lemma therefore makes
$j\circ i$ a closed group embedding. As $P$ was arbitrary, this is
$\mathcal U_p(G)$.
\end{proof}

For $G=J_p$, the first hypothesis is supplied by Camina's Theorem~5.
Her Theorem~6, attributed to Lubotzky and Wilson, supplies a single
two-generator pro-$p$ group containing all countably based pro-$p$
groups, and hence is stronger than the second hypothesis above.
The theorem just proved isolates the composition and closed-image
argument in Corollary~2; it does not prove either of those two input
theorems. In particular, the finite-group realization and depth-control
arguments used to establish Theorem~5 are separate mathematical
obligations. Continuity is an explicit hypothesis of both maps in the
reduction; no assertion about automatic continuity of abstract
homomorphisms is used here.
